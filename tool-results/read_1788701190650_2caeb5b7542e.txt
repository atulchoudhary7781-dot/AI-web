    63→// Neural Network Background Component
    64→function NeuralNetworkBackground() {
    65→  const canvasRef = useRef<HTMLCanvasElement>(null)
    66→
    67→  useEffect(() => {
    68→    const canvas = canvasRef.current
    69→    if (!canvas) return
    70→
    71→    const ctx = canvas.getContext('2d')
    72→    if (!ctx) return
    73→
    74→    let animationFrameId: number
    75→    let mouseX = 0
    76→    let mouseY = 0
    77→
    78→    const resizeCanvas = () => {
    79→      canvas.width = window.innerWidth
    80→      canvas.height = window.innerHeight
    81→    }
    82→
    83→    resizeCanvas()
    84→    window.addEventListener('resize', resizeCanvas)
    85→
    86→    const handleMouseMove = (e: MouseEvent) => {
    87→      mouseX = e.clientX
    88→      mouseY = e.clientY
    89→    }
    90→    window.addEventListener('mousemove', handleMouseMove)
    91→
    92→    // Nodes for neural network
    93→    interface Node {
    94→      x: number
    95→      y: number
    96→      vx: number
    97→      vy: number
    98→      radius: number
    99→      opacity: number
   100→    }
   101→
   102→    const nodes: Node[] = []
   103→    const nodeCount = Math.min(80, Math.floor((window.innerWidth * window.innerHeight) / 15000))
   104→
   105→    for (let i = 0; i < nodeCount; i++) {
   106→      nodes.push({
   107→        x: Math.random() * canvas.width,
   108→        y: Math.random() * canvas.height,
   109→        vx: (Math.random() - 0.5) * 0.5,
   110→        vy: (Math.random() - 0.5) * 0.5,
   111→        radius: Math.random() * 2 + 1,
   112→        opacity: Math.random() * 0.5 + 0.3
   113→      })
   114→    }
   115→
   116→    const animate = () => {
   117→      ctx.fillStyle = 'rgba(0, 0, 10, 0.1)'
   118→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   119→
   120→      // Update and draw nodes
   121→      nodes.forEach((node, i) => {
   122→        // Mouse attraction
   123→        const dx = mouseX - node.x
   124→        const dy = mouseY - node.y
   125→        const dist = Math.sqrt(dx * dx + dy * dy)
   126→        
   127→        if (dist < 200) {
   128→          node.vx += dx * 0.00005
   129→          node.vy += dy * 0.00005
   130→        }
   131→
   132→        node.x += node.vx
   133→        node.y += node.vy
   134→
   135→        // Bounce off edges
   136→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   137→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   138→
   139→        // Draw connections
   140→        nodes.slice(i + 1).forEach(otherNode => {
   141→          const distance = Math.sqrt(
   142→            Math.pow(node.x - otherNode.x, 2) + Math.pow(node.y - otherNode.y, 2)
   143→          )
   144→          
   145→          if (distance < 150) {
   146→            const opacity = (1 - distance / 150) * 0.3
   147→            ctx.beginPath()
   148→            ctx.strokeStyle = `rgba(0, 245, 255, ${opacity})`
   149→            ctx.lineWidth = 0.5
   150→            ctx.moveTo(node.x, node.y)
   151→            ctx.lineTo(otherNode.x, otherNode.y)
   152→            ctx.stroke()
   153→          }
   154→        })
   155→
   156→        // Draw node
   157→        ctx.beginPath()
   158→        const gradient = ctx.createRadialGradient(
   159→          node.x, node.y, 0,
   160→          node.x, node.y, node.radius * 2
   161→        )
   162→        gradient.addColorStop(0, `rgba(0, 245, 255, ${node.opacity})`)
   163→        gradient.addColorStop(1, 'rgba(124, 58, 237, 0)')
   164→        ctx.fillStyle = gradient
   165→        ctx.arc(node.x, node.y, node.radius * 2, 0, Math.PI * 2)
   166→        ctx.fill()
   167→      })
   168→
   169→      animationFrameId = requestAnimationFrame(animate)
   170→    }
   171→
   172→    animate()
   173→
   174→    return () => {
   175→      window.removeEventListener('resize', resizeCanvas)
   176→      window.removeEventListener('mousemove', handleMouseMove)
   177→      cancelAnimationFrame(animationFrameId)
   178→    }
   179→  }, [])
   180→
   181→  return (
   182→    <canvas
   183→      ref={canvasRef}
   184→      className="fixed inset-0 pointer-events-none z-0"
   185→      style={{ background: '#00000a' }}
   186→    />
   187→  )
   188→}

   190→// Glitch Text Component
   191→function GlitchText({ text, className = '' }: { text: string; className?: string }) {
   192→  const [glitchActive, setGlitchActive] = useState(false)
   193→
   194→  useEffect(() => {
   195→    const interval = setInterval(() => {
   196→      setGlitchActive(true)
   197→      setTimeout(() => setGlitchActive(false), 200)
   198→    }, 3000)
   199→    return () => clearInterval(interval)
   200→  }, [])
   201→
   202→  return (
   203→    <span className={`relative inline-block ${className}`}>
   204→      <span className={`relative z-10 ${glitchActive ? 'animate-pulse' : ''}`}>
   205→        {text}
   206→      </span>
   207→      {glitchActive && (
   208→        <>
   209→          <span className="absolute top-0 left-0.5 text-[#ff00aa] opacity-80 clip-text-glitch-1" aria-hidden="true">
   210→            {text}
   211→          </span>
   212→          <span className="absolute top-0 -left-0.5 text-[#00f5ff] opacity-80 clip-text-glitch-2" aria-hidden="true">
   213→            {text}
   214→          </span>
   215→        </>
   216→      )}

   221→// Animated Counter Component
   222→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
   223→  const [count, setCount] = useState(0)
   224→  const [isVisible, setIsVisible] = useState(false)
   225→  const ref = useRef<HTMLDivElement>(null)
   226→
   227→  useEffect(() => {
   228→    const observer = new IntersectionObserver(
   229→      ([entry]) => {
   230→        if (entry.isIntersecting && !isVisible) {
   231→          setIsVisible(true)
   232→        }
   233→      },
   234→      { threshold: 0.1 }
   235→    )
   236→
   237→    if (ref.current) observer.observe(ref.current)
   238→    return () => observer.disconnect()
   239→  }, [isVisible])
   240→
   241→  useEffect(() => {
   242→    if (!isVisible) return
   243→    
   244→    let start = 0
   245→    const increment = target / (duration / 16)
   246→    
   247→    const timer = setInterval(() => {
   248→      start += increment
   249→      if (start >= target) {
   250→        setCount(target)
   251→        clearInterval(timer)
   252→      } else {
   253→        setCount(Math.floor(start))
   254→      }
   255→    }, 16)
   256→
   257→    return () => clearInterval(timer)
   258→  }, [isVisible, target, duration])
   259→
   260→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
   261→}
