     1→'use client'
     2→
     3→import { useState, useEffect, useRef } from 'react'
     4→import { Send, RefreshCw, Copy, Check, Sparkles, User, Bot, ThumbsUp, ThumbsDown, RotateCcw, Square, Paperclip, X } from 'lucide-react'
     5→import { Button } from '@/components/ui/button'
     6→import { Textarea } from '@/components/ui/textarea'
     7→
     8→interface ChatMessage {
     9→  id: string
    10→  role: 'user' | 'assistant'
    11→  content: string
    12→  timestamp: Date

    14→  imageMimeType?: string

    16→  fileName?: string
    17→  fileType?: string
    18→  fileSize?: number
    19→}
    20→
    21→interface FullScreenChatProps {

    24→  setInputValue: (value: string) => void
    25→  isLoading: boolean
    26→  onSubmit: () => void

    29→  onCopy: (text: string) => void
    30→  onFileAttach?: (file: File) => void
    31→  onClearAttachment?: () => void // Callback to clear attachment after send
    32→}
    33→
    34→export default function FullScreenChat({
    35→  messages,
    36→  inputValue,
    37→  setInputValue,
    38→  isLoading,
    39→  onSubmit,
    40→  onStop,
    41→  copiedCode,
    42→  onCopy,
    43→  onFileAttach,
    44→  onClearAttachment
    45→}: FullScreenChatProps) {
    46→  const chatEndRef = useRef<HTMLDivElement>(null)
    47→  const textareaRef = useRef<HTMLTextAreaElement>(null)

    49→  const [attachedFile, setAttachedFile] = useState<File | null>(null)
    50→  const [filePreview, setFilePreview] = useState<string | null>(null)
    51→
    52→  // Sync with parent - clear file when parent sends
    53→  useEffect(() => {
    54→    if (!isLoading && attachedFile && onClearAttachment) {
    55→      // File will be cleared by parent after submit
    56→    }
    57→  }, [isLoading])
    58→

    63→

    69→      onSubmit()
    70→    }
    71→  }

    76→    if (file) {
    77→      setAttachedFile(file)
    78→      
    79→      // Create preview for images
    80→      if (file.type.startsWith('image/')) {
    81→        const reader = new FileReader()
    82→        reader.onloadend = () => {
    83→          setFilePreview(reader.result as string)
    84→        }
    85→        reader.readAsDataURL(file)
    86→      } else {
    87→        setFilePreview(null)
    88→      }
    89→      
    90→      if (onFileAttach) {
    91→        onFileAttach(file)
    92→      }
    93→    }
    94→  }
    95→
    96→  // Handle file attachment click
    97→  const handleAttachClick = () => {
    98→    fileInputRef.current?.click()
    99→  }

   101→  // Remove attached file

   104→    setFilePreview(null)

   136→        <div className="flex-1 flex items-center justify-center px-4 overflow-y-auto custom-scrollbar sidebar-scroll">
   137→          <div className="text-center max-w-2xl py-8">
   138→            {/* Logo */}
   139→            <div className="mb-6 inline-flex items-center justify-center w-16 h-16 rounded-2xl bg-gradient-to-br from-cyan-500 to-violet-600 shadow-2xl shadow-cyan-500/25 animate-pulse-slow">
   140→              <Sparkles className="w-8 h-8 text-white" />

   142→            
   143→            <h2 className="text-2xl sm:text-3xl font-bold text-white mb-3 font-[family-name:var(--font-orbitron)]">
   144→              Welcome to{' '}
   145→              <span className="bg-gradient-to-r from-cyan-400 via-violet-400 to-pink-400 bg-clip-text text-transparent">
   146→                NEXUS AI
   147→              </span>
   148→            </h2>
   149→            
   150→            <p className="text-gray-400 text-base mb-6 max-w-xl mx-auto leading-relaxed">
   151→              Your advanced AI assistant powered by Llama 3.1. Ask me anything — I'm here to help!
   152→            </p>
   153→
   154→            {/* Quick action buttons */}
   155→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 max-w-lg mx-auto">
   156→              {[
   157→                { icon: '💡', text: 'Explain', prompt: 'Explain quantum computing' },
   158→                { icon: '💻', text: 'Code', prompt: 'Write Python code' },
   159→                { icon: '📝', text: 'Write', prompt: 'Help me write an email' },
   160→                { icon: '🎯', text: 'Ideas', prompt: 'Creative business ideas' }

   162→                <button
   163→                  key={i}
   164→                  onClick={() => setInputValue(action.prompt)}
   165→                  className="flex flex-col items-center gap-1.5 p-3 rounded-xl bg-gray-800/50 border border-gray-700/50 hover:border-cyan-500/30 hover:bg-gray-800 transition-all duration-200 group"
   166→                >
   167→                  <span className="text-lg">{action.icon}</span>
   168→                  <span className="text-xs text-gray-300 group-hover:text-white transition-colors">{action.text}</span>

   185→              hasImage: !!message.image,
   186→              content: message.content?.substring(0, 50),
   187→              shouldShowDoc: (message.fileName || (message.content && message.content.includes('📎'))) && !message.image
   188→            })
   189→            
   190→            return (
   191→            <div
   192→              key={message.id}
   193→              className={`flex gap-4 ${message.role === 'user' ? 'justify-end' : ''} animate-fadeIn`}
   194→            >
   195→              {message.role === 'assistant' && (
   196→                <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-cyan-500 to-violet-600 flex items-center justify-center flex-shrink-0 shadow-lg shadow-cyan-500/20">
   197→                  <Sparkles className="w-5 h-5 text-white" />

   200→              
   201→              <div className={`relative group ${message.role === 'user' ? 'order-1 max-w-[85%] md:max-w-[75%]' : 'max-w-[85%] md:max-width-[80%]'}`}>
   202→                {/* Message Bubble */}

   209→                  {message.role === 'user' && (
   210→                    <div className="absolute inset-0 bg-gradient-to-r from-transparent via-white/5 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300" />

   213→                  {/* Image Display - Show attached image in message */}
   214→                  {message.image && (
   215→                    <div className="mb-3 rounded-xl overflow-hidden border border-white/20 shadow-lg cursor-pointer hover:opacity-90 transition-opacity">

   227→                        }}
   228→                      />
   229→                    </div>

   231→
   232→                  {/* Document Display - Show attached document (PDF, DOC, etc.) */}
   233→                  {(message.fileName || (message.content && message.content.includes('📎'))) && !message.image && (
   234→                    <div className={`mb-3 p-4 rounded-xl border-2 ${

   255→                            <span className="text-green-400 font-bold text-sm">XLS</span>
   256→                          ) : message.fileType?.includes('text') || message.fileName?.endsWith('.txt') ? (

   259→                            <span className="text-purple-400 font-bold text-sm">JSON</span>
   260→                          ) : message.fileName?.endsWith('.csv') ? (
   261→                            <span className="text-emerald-400 font-bold text-sm">CSV</span>
   262→                          ) : (
   263→                            <Paperclip className="w-7 h-7 text-cyan-400" />
   264→                          )}
   265→                        </div>
   266→                        
   267→                        {/* File Info */}
   268→                        <div className="flex-1 min-w-0">
   269→                          <p className="text-base font-semibold truncate text-white">

   331→              {message.role === 'user' && (

   557→// Render markdown-like content
   558→function renderMessageContent(
   559→  content: string, 
   560→  onCopy: (text: string) => void, 
   561→  copiedCode: boolean
   562→): React.ReactNode {
   563→  const lines = content.split('\n')
   564→  const elements: React.ReactNode[] = []
   565→  let currentParagraph = ''
   566→  let inCodeBlock = false
   567→  let codeContent = ''
   568→
   569→  const flushParagraph = () => {
   570→    if (currentParagraph.trim()) {
   571→      elements.push(
   572→        <p key={`p-${elements.length}`} className="text-gray-200 my-2 whitespace-pre-wrap leading-relaxed">
   573→          {formatInlineMarkdown(currentParagraph.trim())}
   574→        </p>
   575→      )
   576→      currentParagraph = ''
   577→    }
   578→  }
   579→
   580→  lines.forEach((line, lineIndex) => {
   581→    // Code block handling
   582→    if (line.trim().startsWith('```')) {
   583→      if (inCodeBlock) {
   584→        elements.push(
   585→          <div key={`code-${elements.length}`} className="my-4 relative group/code">
   586→            {/* Code header */}
   587→            <div className="flex items-center justify-between bg-gray-900 rounded-t-lg px-4 py-2 border-b border-gray-700">
   588→              <span className="text-xs text-gray-400 font-mono">code</span>
   589→              <button
   590→                onClick={() => onCopy(codeContent)}
   591→                className="flex items-center gap-1.5 px-2.5 py-1 bg-gray-800 hover:bg-gray-700 rounded-md text-xs text-gray-300 transition-colors opacity-0 group-hover/code:opacity-100"
   592→              >
   593→                {copiedCode ? (
   594→                  <>
   595→                    <Check className="w-3 h-3 text-green-400" />
   596→                    Copied!
   597→                  </>
   598→                ) : (
   599→                  <>
   600→                    <Copy className="w-3 h-3" />
   601→                    Copy
   602→                  </>
   603→                )}
   604→              </button>
   605→            </div>
   606→            <pre className="bg-gray-900/80 rounded-b-lg p-4 overflow-x-auto text-sm border border-gray-800 border-t-0">
   607→              <code className="text-emerald-400 font-mono leading-relaxed">{codeContent}</code>
   608→            </pre>
   609→          </div>
   610→        )
   611→        codeContent = ''
   612→        inCodeBlock = false
   613→      } else {
   614→        flushParagraph()
   615→        inCodeBlock = true
   616→      }
   617→      return
   618→    }
   619→
   620→    if (inCodeBlock) {
   621→      codeContent += line + '\n'
   622→      return
   623→    }
   624→
   625→    // Headers
   626→    if (line.startsWith('# ')) {
   627→      flushParagraph()
   628→      elements.push(<h1 key={lineIndex} className="text-2xl font-bold text-white mt-6 mb-3 text-gradient">{line.slice(2)}</h1>)
   629→      return
   630→    }
   631→    if (line.startsWith('## ')) {
   632→      flushParagraph()
   633→      elements.push(<h2 key={lineIndex} className="text-xl font-semibold text-white mt-5 mb-2.5">{line.slice(3)}</h2>)
   634→      return
   635→    }
   636→    if (line.startsWith('### ')) {
   637→      flushParagraph()
   638→      elements.push(<h3 key={lineIndex} className="text-lg font-medium text-cyan-400 mt-4 mb-2">{line.slice(4)}</h3>)
   639→      return
   640→    }
   641→
   642→    // Lists
   643→    if (line.match(/^\s*[-*]\s+/)) {
   644→      flushParagraph()
   645→      elements.push(
   646→        <li key={lineIndex} className="ml-4 text-gray-200 list-disc mb-1 marker:text-cyan-500">
   647→          {formatInlineMarkdown(line.replace(/^\s*[-*]\s+/, ''))}
   648→        </li>
   649→      )
   650→      return
   651→    }
   652→    if (line.match(/^\d+\.\s+/)) {
   653→      flushParagraph()
   654→      elements.push(
   655→        <li key={lineIndex} className="ml-4 text-gray-200 list-decimal mb-1 marker:text-violet-400">
   656→          {formatInlineMarkdown(line.replace(/^\d+\.\s+/, ''))}
   657→        </li>
   658→      )
   659→      return
   660→    }
   661→
   662→    // Horizontal rule
   663→    if (line.trim() === '---') {
   664→      flushParagraph()
   665→      elements.push(<hr key={lineIndex} className="my-4 border-gray-700/50" />)
   666→      return
   667→    }
   668→
   669→    // Regular paragraph
   670→    if (line.trim()) {
   671→      currentParagraph += (currentParagraph ? ' ' : '') + line
   672→    } else {
   673→      flushParagraph()
   674→    }
   675→  })
   676→
   677→  flushParagraph()
   678→
   679→  return elements.length > 0 ? elements : content
   680→}

   682→// Format inline markdown
   683→function formatInlineMarkdown(text: string): React.ReactNode {
   684→  const parts: React.ReactNode[] = []
   685→  const regex = /\*\*(.*?)\*\*|`(.*?)`|\*(.*?)\*/g
   686→  let lastIndex = 0
   687→  let match
   688→  
   689→  while ((match = regex.exec(text)) !== null) {
   690→    if (match.index > lastIndex) {
   691→      parts.push(text.slice(lastIndex, match.index))
   692→    }
   693→    
   694→    if (match[1]) {
   695→      parts.push(<strong key={match.index} className="font-semibold text-white">{match[1]}</strong>)
   696→    } else if (match[2]) {
   697→      parts.push(<code key={match.index} className="px-2 py-0.5 bg-gray-800/80 rounded-md text-cyan-300 text-sm font-mono border border-gray-700/50">{match[2]}</code>)
   698→    } else if (match[3]) {
   699→      parts.push(<em key={match.index} className="italic text-gray-300">{match[3]}</em>)
   700→    }
   701→    
   702→    lastIndex = regex.lastIndex
   703→  }
   704→  
   705→  if (lastIndex < text.length) {
   706→    parts.push(text.slice(lastIndex))
   707→  }
   708→  
   709→  return parts.length > 0 ? parts : text
   710→}
