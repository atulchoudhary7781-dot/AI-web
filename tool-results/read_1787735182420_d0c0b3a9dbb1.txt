   580→// Render markdown-like content
   581→function renderMessageContent(
   582→  content: string, 
   583→  onCopy: (text: string) => void, 
   584→  copiedCode: boolean
   585→): React.ReactNode {
   586→  const lines = content.split('\n')
   587→  const elements: React.ReactNode[] = []
   588→  let currentParagraph = ''
   589→  let inCodeBlock = false
   590→  let codeContent = ''
   591→
   592→  const flushParagraph = () => {
   593→    if (currentParagraph.trim()) {
   594→      elements.push(
   595→        <p key={`p-${elements.length}`} className="text-gray-200 my-2 whitespace-pre-wrap leading-relaxed">
   596→          {formatInlineMarkdown(currentParagraph.trim())}
   597→        </p>
   598→      )
   599→      currentParagraph = ''
   600→    }
   601→  }
   602→
   603→  lines.forEach((line, lineIndex) => {
   604→    // Code block handling
   605→    if (line.trim().startsWith('```')) {
   606→      if (inCodeBlock) {
   607→        elements.push(
   608→          <div key={`code-${elements.length}`} className="my-4 relative group/code">
   609→            {/* Code header */}
   610→            <div className="flex items-center justify-between bg-gray-900 rounded-t-lg px-4 py-2 border-b border-gray-700">
   611→              <span className="text-xs text-gray-400 font-mono">code</span>
   612→              <button
   613→                onClick={() => onCopy(codeContent)}
   614→                className="flex items-center gap-1.5 px-2.5 py-1 bg-gray-800 hover:bg-gray-700 rounded-md text-xs text-gray-300 transition-colors opacity-0 group-hover/code:opacity-100"
   615→              >
   616→                {copiedCode ? (
   617→                  <>
   618→                    <Check className="w-3 h-3 text-green-400" />
   619→                    Copied!
   620→                  </>
   621→                ) : (
   622→                  <>
   623→                    <Copy className="w-3 h-3" />
   624→                    Copy
   625→                  </>
   626→                )}
   627→              </button>
   628→            </div>
   629→            <pre className="bg-gray-900/80 rounded-b-lg p-4 overflow-x-auto text-sm border border-gray-800 border-t-0">
   630→              <code className="text-emerald-400 font-mono leading-relaxed">{codeContent}</code>
   631→            </pre>
   632→          </div>
   633→        )
   634→        codeContent = ''
   635→        inCodeBlock = false
   636→      } else {
   637→        flushParagraph()
   638→        inCodeBlock = true
   639→      }
   640→      return
   641→    }
   642→
   643→    if (inCodeBlock) {
   644→      codeContent += line + '\n'
   645→      return
   646→    }
   647→
   648→    // Headers
   649→    if (line.startsWith('# ')) {
   650→      flushParagraph()
   651→      elements.push(<h1 key={lineIndex} className="text-2xl font-bold text-white mt-6 mb-3 text-gradient">{line.slice(2)}</h1>)
   652→      return
   653→    }
   654→    if (line.startsWith('## ')) {
   655→      flushParagraph()
   656→      elements.push(<h2 key={lineIndex} className="text-xl font-semibold text-white mt-5 mb-2.5">{line.slice(3)}</h2>)
   657→      return
   658→    }
   659→    if (line.startsWith('### ')) {
   660→      flushParagraph()
   661→      elements.push(<h3 key={lineIndex} className="text-lg font-medium text-cyan-400 mt-4 mb-2">{line.slice(4)}</h3>)
   662→      return
   663→    }
   664→
   665→    // Lists
   666→    if (line.match(/^\s*[-*]\s+/)) {
   667→      flushParagraph()
   668→      elements.push(
   669→        <li key={lineIndex} className="ml-4 text-gray-200 list-disc mb-1 marker:text-cyan-500">
   670→          {formatInlineMarkdown(line.replace(/^\s*[-*]\s+/, ''))}
   671→        </li>
   672→      )
   673→      return
   674→    }
   675→    if (line.match(/^\d+\.\s+/)) {
   676→      flushParagraph()
   677→      elements.push(
   678→        <li key={lineIndex} className="ml-4 text-gray-200 list-decimal mb-1 marker:text-violet-400">
   679→          {formatInlineMarkdown(line.replace(/^\d+\.\s+/, ''))}
   680→        </li>
   681→      )
   682→      return
   683→    }
   684→
   685→    // Horizontal rule
   686→    if (line.trim() === '---') {
   687→      flushParagraph()
   688→      elements.push(<hr key={lineIndex} className="my-4 border-gray-700/50" />)
   689→      return
   690→    }
   691→
   692→    // Regular paragraph
   693→    if (line.trim()) {
   694→      currentParagraph += (currentParagraph ? ' ' : '') + line
   695→    } else {
   696→      flushParagraph()
   697→    }
   698→  })
   699→
   700→  flushParagraph()
   701→
   702→  return elements.length > 0 ? elements : content
   703→}

   705→// Format inline markdown
   706→function formatInlineMarkdown(text: string): React.ReactNode {
   707→  const parts: React.ReactNode[] = []
   708→  const regex = /\*\*(.*?)\*\*|`(.*?)`|\*(.*?)\*/g
   709→  let lastIndex = 0
   710→  let match
   711→  
   712→  while ((match = regex.exec(text)) !== null) {
   713→    if (match.index > lastIndex) {
   714→      parts.push(text.slice(lastIndex, match.index))
   715→    }
   716→    
   717→    if (match[1]) {
   718→      parts.push(<strong key={match.index} className="font-semibold text-white">{match[1]}</strong>)
   719→    } else if (match[2]) {
   720→      parts.push(<code key={match.index} className="px-2 py-0.5 bg-gray-800/80 rounded-md text-cyan-300 text-sm font-mono border border-gray-700/50">{match[2]}</code>)
   721→    } else if (match[3]) {
   722→      parts.push(<em key={match.index} className="italic text-gray-300">{match[3]}</em>)
   723→    }
   724→    
   725→    lastIndex = regex.lastIndex
   726→  }
   727→  
   728→  if (lastIndex < text.length) {
   729→    parts.push(text.slice(lastIndex))
   730→  }
   731→  
   732→  return parts.length > 0 ? parts : text
   733→}
