    48→// Neural Network Background Component
    49→function NeuralNetworkBackground() {
    50→  const canvasRef = useRef<HTMLCanvasElement>(null)
    51→
    52→  useEffect(() => {
    53→    const canvas = canvasRef.current
    54→    if (!canvas) return
    55→
    56→    const ctx = canvas.getContext('2d')
    57→    if (!ctx) return
    58→
    59→    let animationFrameId: number
    60→    let mouseX = 0
    61→    let mouseY = 0
    62→
    63→    const resizeCanvas = () => {
    64→      canvas.width = window.innerWidth
    65→      canvas.height = window.innerHeight
    66→    }
    67→
    68→    resizeCanvas()
    69→    window.addEventListener('resize', resizeCanvas)
    70→
    71→    const handleMouseMove = (e: MouseEvent) => {
    72→      mouseX = e.clientX
    73→      mouseY = e.clientY
    74→    }
    75→    window.addEventListener('mousemove', handleMouseMove)
    76→
    77→    // Nodes for neural network
    78→    interface Node {
    79→      x: number
    80→      y: number
    81→      vx: number
    82→      vy: number
    83→      radius: number
    84→      opacity: number
    85→    }
    86→
    87→    const nodes: Node[] = []
    88→    const nodeCount = Math.min(80, Math.floor((window.innerWidth * window.innerHeight) / 15000))
    89→
    90→    for (let i = 0; i < nodeCount; i++) {
    91→      nodes.push({
    92→        x: Math.random() * canvas.width,
    93→        y: Math.random() * canvas.height,
    94→        vx: (Math.random() - 0.5) * 0.5,
    95→        vy: (Math.random() - 0.5) * 0.5,
    96→        radius: Math.random() * 2 + 1,
    97→        opacity: Math.random() * 0.5 + 0.3
    98→      })
    99→    }
   100→
   101→    const animate = () => {
   102→      ctx.fillStyle = 'rgba(0, 0, 10, 0.1)'
   103→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   104→
   105→      // Update and draw nodes
   106→      nodes.forEach((node, i) => {
   107→        // Mouse attraction
   108→        const dx = mouseX - node.x
   109→        const dy = mouseY - node.y
   110→        const dist = Math.sqrt(dx * dx + dy * dy)
   111→        
   112→        if (dist < 200) {
   113→          node.vx += dx * 0.00005
   114→          node.vy += dy * 0.00005
   115→        }
   116→
   117→        node.x += node.vx
   118→        node.y += node.vy
   119→
   120→        // Bounce off edges
   121→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   122→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   123→
   124→        // Draw connections
   125→        nodes.slice(i + 1).forEach(otherNode => {
   126→          const distance = Math.sqrt(
   127→            Math.pow(node.x - otherNode.x, 2) + Math.pow(node.y - otherNode.y, 2)
   128→          )
   129→          
   130→          if (distance < 150) {
   131→            const opacity = (1 - distance / 150) * 0.3
   132→            ctx.beginPath()
   133→            ctx.strokeStyle = `rgba(0, 245, 255, ${opacity})`
   134→            ctx.lineWidth = 0.5
   135→            ctx.moveTo(node.x, node.y)
   136→            ctx.lineTo(otherNode.x, otherNode.y)
   137→            ctx.stroke()
   138→          }
   139→        })
   140→
   141→        // Draw node
   142→        ctx.beginPath()
   143→        const gradient = ctx.createRadialGradient(
   144→          node.x, node.y, 0,
   145→          node.x, node.y, node.radius * 2
   146→        )
   147→        gradient.addColorStop(0, `rgba(0, 245, 255, ${node.opacity})`)
   148→        gradient.addColorStop(1, 'rgba(124, 58, 237, 0)')
   149→        ctx.fillStyle = gradient
   150→        ctx.arc(node.x, node.y, node.radius * 2, 0, Math.PI * 2)
   151→        ctx.fill()
   152→      })
   153→
   154→      animationFrameId = requestAnimationFrame(animate)
   155→    }
   156→
   157→    animate()
   158→
   159→    return () => {
   160→      window.removeEventListener('resize', resizeCanvas)
   161→      window.removeEventListener('mousemove', handleMouseMove)
   162→      cancelAnimationFrame(animationFrameId)
   163→    }
   164→  }, [])
   165→
   166→  return (
   167→    <canvas
   168→      ref={canvasRef}
   169→      className="fixed inset-0 pointer-events-none z-0"
   170→      style={{ background: '#00000a' }}
   171→    />
   172→  )
   173→}

   175→// Glitch Text Component
   176→function GlitchText({ text, className = '' }: { text: string; className?: string }) {
   177→  const [glitchActive, setGlitchActive] = useState(false)
   178→
   179→  useEffect(() => {
   180→    const interval = setInterval(() => {
   181→      setGlitchActive(true)
   182→      setTimeout(() => setGlitchActive(false), 200)
   183→    }, 3000)
   184→    return () => clearInterval(interval)
   185→  }, [])
   186→
   187→  return (
   188→    <span className={`relative inline-block ${className}`}>
   189→      <span className={`relative z-10 ${glitchActive ? 'animate-pulse' : ''}`}>
   190→        {text}
   191→      </span>
   192→      {glitchActive && (
   193→        <>
   194→          <span className="absolute top-0 left-0.5 text-[#ff00aa] opacity-80 clip-text-glitch-1" aria-hidden="true">
   195→            {text}
   196→          </span>
   197→          <span className="absolute top-0 -left-0.5 text-[#00f5ff] opacity-80 clip-text-glitch-2" aria-hidden="true">
   198→            {text}
   199→          </span>
   200→        </>
   201→      )}
   202→    </span>
   203→  )
   204→}

   206→// Animated Counter Component
   207→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
   208→  const [count, setCount] = useState(0)
   209→  const [isVisible, setIsVisible] = useState(false)
   210→  const ref = useRef<HTMLDivElement>(null)
   211→
   212→  useEffect(() => {
   213→    const observer = new IntersectionObserver(
   214→      ([entry]) => {
   215→        if (entry.isIntersecting && !isVisible) {
   216→          setIsVisible(true)
   217→        }
   218→      },
   219→      { threshold: 0.1 }
   220→    )
   221→
   222→    if (ref.current) observer.observe(ref.current)
   223→    return () => observer.disconnect()
   224→  }, [isVisible])
   225→
   226→  useEffect(() => {
   227→    if (!isVisible) return
   228→    
   229→    let start = 0
   230→    const increment = target / (duration / 16)
   231→    
   232→    const timer = setInterval(() => {
   233→      start += increment
   234→      if (start >= target) {
   235→        setCount(target)
   236→        clearInterval(timer)
   237→      } else {
   238→        setCount(Math.floor(start))
   239→      }
   240→    }, 16)
   241→
   242→    return () => clearInterval(timer)
   243→  }, [isVisible, target, duration])
   244→
   245→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
   246→}
