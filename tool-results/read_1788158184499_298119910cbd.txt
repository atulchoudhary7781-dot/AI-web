    37→    37→// Neural Network Background Component
    38→    38→function NeuralNetworkBackground() {
    39→    39→  const canvasRef = useRef<HTMLCanvasElement>(null)
    40→    40→
    41→    41→  useEffect(() => {
    42→    42→    const canvas = canvasRef.current
    43→    43→    if (!canvas) return
    44→    44→
    45→    45→    const ctx = canvas.getContext('2d')
    46→    46→    if (!ctx) return
    47→    47→
    48→    48→    let animationFrameId: number
    49→    49→    let mouseX = 0
    50→    50→    let mouseY = 0
    51→    51→
    52→    52→    const resizeCanvas = () => {
    53→    53→      canvas.width = window.innerWidth
    54→    54→      canvas.height = window.innerHeight
    55→    55→    }
    56→    56→
    57→    57→    resizeCanvas()
    58→    58→    window.addEventListener('resize', resizeCanvas)
    59→    59→
    60→    60→    const handleMouseMove = (e: MouseEvent) => {
    61→    61→      mouseX = e.clientX
    62→    62→      mouseY = e.clientY
    63→    63→    }
    64→    64→    window.addEventListener('mousemove', handleMouseMove)
    65→    65→
    66→    66→    // Nodes for neural network
    67→    67→    interface Node {
    68→    68→      x: number
    69→    69→      y: number
    70→    70→      vx: number
    71→    71→      vy: number
    72→    72→      radius: number
    73→    73→      opacity: number
    74→    74→    }
    75→    75→
    76→    76→    const nodes: Node[] = []
    77→    77→    const nodeCount = Math.min(80, Math.floor((window.innerWidth * window.innerHeight) / 15000))
    78→    78→
    79→    79→    for (let i = 0; i < nodeCount; i++) {
    80→    80→      nodes.push({
    81→    81→        x: Math.random() * canvas.width,
    82→    82→        y: Math.random() * canvas.height,
    83→    83→        vx: (Math.random() - 0.5) * 0.5,
    84→    84→        vy: (Math.random() - 0.5) * 0.5,
    85→    85→        radius: Math.random() * 2 + 1,
    86→    86→        opacity: Math.random() * 0.5 + 0.3
    87→    87→      })
    88→    88→    }
    89→    89→
    90→    90→    const animate = () => {
    91→    91→      ctx.fillStyle = 'rgba(0, 0, 10, 0.1)'
    92→    92→      ctx.fillRect(0, 0, canvas.width, canvas.height)
    93→    93→
    94→    94→      // Update and draw nodes
    95→    95→      nodes.forEach((node, i) => {
    96→    96→        // Mouse attraction
    97→    97→        const dx = mouseX - node.x
    98→    98→        const dy = mouseY - node.y
    99→    99→        const dist = Math.sqrt(dx * dx + dy * dy)
   100→   100→        
   101→   101→        if (dist < 200) {
   102→   102→          node.vx += dx * 0.00005
   103→   103→          node.vy += dy * 0.00005
   104→   104→        }
   105→   105→
   106→   106→        node.x += node.vx
   107→   107→        node.y += node.vy
   108→   108→
   109→   109→        // Bounce off edges
   110→   110→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   111→   111→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   112→   112→
   113→   113→        // Draw connections
   114→   114→        nodes.slice(i + 1).forEach(otherNode => {
   115→   115→          const distance = Math.sqrt(
   116→   116→            Math.pow(node.x - otherNode.x, 2) + Math.pow(node.y - otherNode.y, 2)
   117→   117→          )
   118→   118→          
   119→   119→          if (distance < 150) {
   120→   120→            const opacity = (1 - distance / 150) * 0.3
   121→   121→            ctx.beginPath()
   122→   122→            ctx.strokeStyle = `rgba(0, 245, 255, ${opacity})`
   123→   123→            ctx.lineWidth = 0.5
   124→   124→            ctx.moveTo(node.x, node.y)
   125→   125→            ctx.lineTo(otherNode.x, otherNode.y)
   126→   126→            ctx.stroke()
   127→   127→          }
   128→   128→        })
   129→   129→
   130→   130→        // Draw node
   131→   131→        ctx.beginPath()
   132→   132→        const gradient = ctx.createRadialGradient(
   133→   133→          node.x, node.y, 0,
   134→   134→          node.x, node.y, node.radius * 2
   135→   135→        )
   136→   136→        gradient.addColorStop(0, `rgba(0, 245, 255, ${node.opacity})`)
   137→   137→        gradient.addColorStop(1, 'rgba(124, 58, 237, 0)')
   138→   138→        ctx.fillStyle = gradient
   139→   139→        ctx.arc(node.x, node.y, node.radius * 2, 0, Math.PI * 2)
   140→   140→        ctx.fill()
   141→   141→      })
   142→   142→
   143→   143→      animationFrameId = requestAnimationFrame(animate)
   144→   144→    }
   145→   145→
   146→   146→    animate()
   147→   147→
   148→   148→    return () => {
   149→   149→      window.removeEventListener('resize', resizeCanvas)
   150→   150→      window.removeEventListener('mousemove', handleMouseMove)
   151→   151→      cancelAnimationFrame(animationFrameId)
   152→   152→    }
   153→   153→  }, [])
   154→   154→
   155→   155→  return (
   156→   156→    <canvas
   157→   157→      ref={canvasRef}
   158→   158→      className="fixed inset-0 pointer-events-none z-0"
   159→   159→      style={{ background: '#00000a' }}
   160→   160→    />
   161→   161→  )
   162→   162→}

   164→   164→// Glitch Text Component
   165→   165→function GlitchText({ text, className = '' }: { text: string; className?: string }) {
   166→   166→  const [glitchActive, setGlitchActive] = useState(false)
   167→   167→
   168→   168→  useEffect(() => {
   169→   169→    const interval = setInterval(() => {
   170→   170→      setGlitchActive(true)

   173→   173→    return () => clearInterval(interval)
   174→   174→  }, [])
   175→   175→
   176→   176→  return (
   177→   177→    <span className={`relative inline-block ${className}`}>
   178→   178→      <span className={`relative z-10 ${glitchActive ? 'animate-pulse' : ''}`}>
   179→   179→        {text}

   181→   181→      {glitchActive && (
   182→   182→        <>
   183→   183→          <span className="absolute top-0 left-0.5 text-[#ff00aa] opacity-80 clip-text-glitch-1" aria-hidden="true">
   184→   184→            {text}

   195→   195→// Animated Counter Component
   196→   196→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
   197→   197→  const [count, setCount] = useState(0)
   198→   198→  const [isVisible, setIsVisible] = useState(false)
   199→   199→  const ref = useRef<HTMLDivElement>(null)
   200→   200→
   201→   201→  useEffect(() => {
   202→   202→    const observer = new IntersectionObserver(
   203→   203→      ([entry]) => {
   204→   204→        if (entry.isIntersecting && !isVisible) {
   205→   205→          setIsVisible(true)
   206→   206→        }
   207→   207→      },
   208→   208→      { threshold: 0.1 }
   209→   209→    )
   210→   210→
   211→   211→    if (ref.current) observer.observe(ref.current)
   212→   212→    return () => observer.disconnect()
   213→   213→  }, [isVisible])
   214→   214→
   215→   215→  useEffect(() => {
   216→   216→    if (!isVisible) return
   217→   217→    
   218→   218→    let start = 0
   219→   219→    const increment = target / (duration / 16)
   220→   220→    
   221→   221→    const timer = setInterval(() => {
   222→   222→      start += increment
   223→   223→      if (start >= target) {
   224→   224→        setCount(target)
   225→   225→        clearInterval(timer)
   226→   226→      } else {
   227→   227→        setCount(Math.floor(start))
   228→   228→      }
   229→   229→    }, 16)
   230→   230→
   231→   231→    return () => clearInterval(timer)
   232→   232→  }, [isVisible, target, duration])
   233→   233→
   234→   234→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
   235→   235→}
