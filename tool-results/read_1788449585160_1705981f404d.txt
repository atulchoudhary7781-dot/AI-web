   112→export function ChatInterface() {
   113→  const { t } = useI18n()

   115→  const [input, setInput] = useState('')
   116→  const [isTyping, setIsTyping] = useState(false)
   117→  const [copiedId, setCopiedId] = useState<string | null>(null)
   118→  const [isExpanded, setIsExpanded] = useState(false)
   119→  
   120→  // Feature C: Chat History state
   121→  const [showHistory, setShowHistory] = useState(false)
   122→  const [currentConversation, setCurrentConversation] = useState<ChatConversation | null>(null)
   123→  
   124→  // AI Model Selector State
   125→  const [selectedModel, setSelectedModel] = useState<AIModel>(AI_MODELS[0])
   126→  const [showModelSelector, setShowModelSelector] = useState(false)
   127→  
   128→  // Reaction states - using simple object
   129→  const [likedMessages, setLikedMessages] = useState<Set<string>>(new Set())
   130→  const [dislikedMessages, setDislikedMessages] = useState<Set<string>>(new Set())
   131→  const [regeneratingId, setRegeneratingId] = useState<string | null>(null)
   132→  
   133→  const messagesEndRef = useRef<HTMLDivElement>(null)
   134→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   135→
   136→  // Auto-scroll to bottom
   137→  const scrollToBottom = () => {
   138→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   139→  }
   140→
   141→  useEffect(() => {
   142→    scrollToBottom()
   143→  }, [messages])
   144→
   145→  // Handle voice input transcript - Feature D
   146→  const handleVoiceTranscript = useCallback((transcript: string) => {
   147→    setInput(prev => prev ? `${prev} ${transcript}` : transcript)
   148→    textareaRef.current?.focus()
   149→  }, [])
   150→
   151→  // Handle send message
   152→  const handleSend = async () => {
   153→    if (!input.trim()) return
   154→
   155→    const userMessage: Message = {
   156→      id: 'user-' + Date.now(),
   157→      role: 'user',
   158→      content: input.trim(),
   159→      timestamp: new Date(),
   160→    }
   161→
   162→    setMessages((prev) => [...prev, userMessage])
   163→    
   164→    setInput('')
   165→    setIsTyping(true)
   166→
   167→    // Simulate AI response
   168→    setTimeout(() => {
   169→      const aiMessageId = 'ai-' + Date.now()
   170→      const aiMessage: Message = {
   171→        id: aiMessageId,
   172→        role: 'assistant',
   173→        content: `🤖 **${selectedModel.name}** (${selectedModel.provider})\n\nI've analyzed your query about "${input.slice(0, 30)}...". Based on my neural processing capabilities using ${selectedModel.name}, I can provide you with comprehensive insights. The data suggests multiple pathways for exploration. Would you like me to elaborate on any specific aspect?`,
   174→        timestamp: new Date(),
   175→        showButtons: false, // Initially HIDE buttons
   176→      }
   177→      
   178→      // Add message first (without buttons)
   179→      setMessages((prev) => [...prev, aiMessage])
   180→      setIsTyping(false)
   181→      
   182→      // Then show buttons after 800ms delay (answer complete effect)
   183→      setTimeout(() => {
   184→        setMessages(prev => prev.map(msg => 
   185→          msg.id === aiMessageId ? { ...msg, showButtons: true } : msg
   186→        ))
   187→      }, 800)
   188→    }, 1500 + Math.random() * 1000)
   189→  }
   190→
   191→  // Handle copy message
   192→  const handleCopy = async (content: string, id: string) => {
   193→    try {
   194→      await navigator.clipboard.writeText(content)
   195→      setCopiedId(id)
   196→      setTimeout(() => setCopiedId(null), 2000)
   197→    } catch (err) {
   198→      console.error('Failed to copy:', err)
   199→      // Fallback for older browsers
   200→      const textArea = document.createElement('textarea')
   201→      textArea.value = content
   202→      document.body.appendChild(textArea)
   203→      textArea.select()
   204→      document.execCommand('copy')
   205→      document.body.removeChild(textArea)
   206→      setCopiedId(id)
   207→      setTimeout(() => setCopiedId(null), 2000)
   208→    }
   209→  }
   210→
   211→  // 👍 HANDLE LIKE - Simple and direct
   212→  const handleLike = (messageId: string) => {
   213→    console.log('👍 LIKE clicked for:', messageId)
   214→    
   215→    // Create new sets to trigger re-render
   216→    const newLiked = new Set(likedMessages)
   217→    const newDisliked = new Set(dislikedMessages)
   218→    
   219→    if (newLiked.has(messageId)) {
   220→      // Already liked - unlike it
   221→      newLiked.delete(messageId)
   222→      console.log('❌ Unliked')
   223→    } else {
   224→      // Like it (and remove dislike if exists)
   225→      newLiked.add(messageId)
   226→      newDisliked.delete(messageId)
   227→      console.log('✅ Liked!')
   228→    }
   229→    
   230→    setLikedMessages(newLiked)
   231→    setDislikedMessages(newDisliked)
   232→  }
   233→
   234→  // 👎 HANDLE DISLIKE - Simple and direct
   235→  const handleDislike = (messageId: string) => {
   236→    console.log('👎 DISLIKE clicked for:', messageId)
   237→    
   238→    // Create new sets to trigger re-render
   239→    const newLiked = new Set(likedMessages)
   240→    const newDisliked = new Set(dislikedMessages)
   241→    
   242→    if (newDisliked.has(messageId)) {
   243→      // Already disliked - undislike it
   244→      newDisliked.delete(messageId)
   245→      console.log('❌ Undisliked')
   246→    } else {
   247→      // Dislike it (and remove like if exists)
   248→      newDisliked.add(messageId)
   249→      newLiked.delete(messageId)
   250→      console.log('✅ Disliked!')
   251→    }
   252→    
   253→    setLikedMessages(newLiked)
   254→    setDislikedMessages(newDisliked)
   255→  }
   256→
   257→  // 🔄 HANDLE REGENERATE - Simple and direct
   258→  const handleRegenerate = (messageId: string) => {
   259→    console.log('🔄 REGENERATE clicked for:', messageId)
   260→    
   261→    // Prevent if already regenerating or typing
   262→    if (isTyping || regeneratingId) {
   263→      console.log('⚠️ Already busy, ignoring')
   264→      return
   265→    }
   266→    
   267→    // Find the AI message index
   268→    const aiIndex = messages.findIndex(m => m.id === messageId)
   269→    console.log('AI Message index:', aiIndex)
   270→    
   271→    if (aiIndex <= 0) {
   272→      console.log('❌ Invalid message or first message')
   273→      return
   274→    }
   275→    
   276→    // Get user message before this AI response
   277→    const userMessage = messages[aiIndex - 1]
   278→    if (!userMessage || userMessage.role !== 'user') {
   279→      console.log('❌ No user message found before')
   280→      return
   281→    }
   282→    
   283→    console.log('📝 Regenerating for:', userMessage.content.slice(0, 50))
   284→    
   285→    // Start regeneration
   286→    setRegeneratingId(messageId)
   287→    setIsTyping(true)
   288→    
   289→    // Remove old AI message
   290→    setMessages(prev => prev.filter(m => m.id !== messageId))
   291→    
   292→    // Generate new response after delay
   293→    setTimeout(() => {
   294→      const newAiId = 'regen-' + Date.now()
   295→      const newAiMessage: Message = {
   296→        id: newAiId,
   297→        role: 'assistant',
   298→        content: `🔄 [REGENERATED with ${selectedModel.name}] I've re-analyzed your query about "${userMessage.content.slice(0, 30)}..." using ${selectedModel.name} (${selectedModel.provider}). Here's an alternative approach based on deeper processing. Does this better address your needs?`,
   299→        timestamp: new Date(),
   300→        showButtons: false, // Initially HIDE buttons
   301→      }
   302→      
   303→      // Add message first (without buttons)
   304→      setMessages(prev => [...prev, newAiMessage])
   305→      setIsTyping(false)
   306→      setRegeneratingId(null)
   307→      
   308→      // Then show buttons after 800ms delay (answer complete effect)
   309→      setTimeout(() => {
   310→        setMessages(prev => prev.map(msg => 
   311→          msg.id === newAiId ? { ...msg, showButtons: true } : msg
   312→        ))
   313→      }, 800)
   314→      
   315→      console.log('✅ Regeneration complete:', newAiId)
   316→    }, 2000)
   317→  }
   318→
   319→  // ↗️ HANDLE SHARE
   320→  const handleShare = async (content: string) => {
   321→    console.log('↗️ SHARE clicked')
   322→    if (navigator.share) {
   323→      try {
   324→        await navigator.share({
   325→          title: 'NEXUS AI Response',
   326→          text: content,
   327→          url: window.location.href
   328→        })
   329→      } catch (err) {
   330→        console.log('Share cancelled or failed')
   331→      }
   332→    } else {
   333→      handleCopy(content, 'share-' + Date.now())
   334→    }
   335→  }
   336→
   337→  // Handle clear chat
   338→  const handleClear = () => {
   339→    setMessages([])
   340→    setLikedMessages(new Set())
   341→    setDislikedMessages(new Set())
   342→  }
   343→
   344→  // Handle key press
   345→  const handleKeyDown = (e: React.KeyboardEvent) => {
   346→    if (e.key === 'Enter' && !e.shiftKey) {
   347→      e.preventDefault()
   348→      handleSend()
   349→    }
   350→  }
   351→
   352→  // Handle conversation selection from history
   353→  const handleSelectConversation = useCallback((conversation: ChatConversation | null) => {
   354→    setCurrentConversation(conversation)
   355→    if (conversation && conversation.messages.length > 0) {
   356→      setMessages(conversation.messages.map(msg => ({
   357→        ...msg,
   358→        timestamp: new Date(msg.timestamp),
   359→        showButtons: msg.role === 'assistant', // Show buttons for AI messages
   360→      })))
   361→    }
   362→    setShowHistory(false)
   363→  }, [])
   364→
   365→  return (
   366→    <div className="flex h-[600px] md:h-[700px] rounded-2xl overflow-hidden transition-all duration-500 glass-strong border border-neon-cyan/20 shadow-[0_0_40px_rgba(0,255,255,0.15)]">
   367→      
   368→      {/* Feature C: Chat History Sidebar */}
   369→      <div 
   370→        className={cn(
   371→          "absolute left-0 top-0 bottom-0 w-80 z-20 transition-transform duration-300 ease-out",
   372→          showHistory ? "translate-x-0" : "-translate-x-full",
   373→          "glass-strong border-r border-neon-cyan/20"
   374→        )}
   375→      >
   376→        <ChatHistory
   377→          onSelectConversation={handleSelectConversation}
   378→          currentConversationId={currentConversation?.id}
   379→          className="h-full"
   380→        />
   381→        
   382→        {/* Close button */}
   383→        <button
   384→          onClick={() => setShowHistory(false)}
   385→          className="absolute top-4 right-4 p-1.5 rounded-lg hover:bg-white/10 transition-colors z-30"
   386→          aria-label="Close history"
   387→        >
   388→          ✕
   389→        </button>
   390→      </div>
   391→
   392→      {/* Main Chat Area */}
   393→      <div className={cn(
   394→        "flex flex-col flex-1 transition-all duration-500",
   395→        isExpanded ? "fixed inset-4 md:inset-8 z-50 rounded-2xl" : "",
   396→        "min-w-0"
   397→      )}>
   398→        {/* Header */}
   399→        <div className="flex items-center justify-between px-6 py-4 border-b border-neon-cyan/20 bg-black/80">
   400→          <div className="flex items-center gap-3">
   401→            {/* History toggle button */}
   402→            <Button
   403→              variant="ghost"
   404→              size="icon-sm"
   405→              onClick={() => setShowHistory(!showHistory)}
   406→              className={cn(
   407→                "text-muted-foreground hover:text-neon-cyan",
   408→                showHistory && "text-neon-cyan bg-neon-cyan/10"
   409→              )}
   410→              aria-label={t('chat.history')}
   411→            >
   412→              <History className="w-4 h-4" />
   413→            </Button>
   414→            
   415→            {/* Status indicator */}
   416→            <div className="relative">
   417→              <div className="w-3 h-3 rounded-full bg-neon-cyan animate-pulse shadow-[0_0_10px_rgba(0,255,255,0.6)]" />
   418→              <div className="absolute inset-0 w-3 h-3 rounded-full bg-neon-cyan animate-ping opacity-30" />
   419→            </div>
   420→            
   421→            <div>
   422→              <h3 className="font-display font-semibold text-foreground flex items-center gap-2">
   423→                <Sparkles className="w-4 h-4 text-neon-cyan" />
   424→                {t('chat.title')}
   425→              </h3>
   426→              <p className="text-xs text-muted-foreground">{t('chat.subtitle')}</p>
   427→            </div>
   428→          </div>
   429→
   430→          <div className="flex items-center gap-2">
   431→            <Button
   432→              variant="ghost"
   433→              size="icon-sm"
   434→              onClick={handleClear}
   435→              className="text-muted-foreground hover:text-red-400"
   436→              aria-label={t('chat.clear')}
   437→            >
   438→              <Trash2 className="w-4 h-4" />
   439→            </Button>
   440→            <Button
   441→              variant="ghost"
   442→              size="icon-sm"
   443→              onClick={() => setIsExpanded(!isExpanded)}
   444→              className="text-muted-foreground hover:text-neon-cyan"
   445→            >
   446→              {isExpanded ? <Minimize2 className="w-4 h-4" /> : <Maximize2 className="w-4 h-4" />}
   447→            </Button>
   448→          </div>
   449→        </div>
   450→
   451→        {/* Messages area */}

   453→          {messages.map((message, index) => (
   454→            <div
   455→              key={message.id}
   456→              className={cn(
   457→                "chat-message-enter flex gap-4",
   458→                message.role === 'user' ? 'flex-row-reverse' : ''
   459→              )}
   460→              style={{ animationDelay: `${index * 0.05}s` }}
   461→            >
   462→              {/* Avatar */}
   463→              <div
   464→                className={cn(
   465→                  "flex-shrink-0 w-9 h-9 rounded-xl flex items-center justify-center",
   466→                  message.role === 'user'
   467→                    ? "bg-gradient-to-br from-neon-cyan to-electric-blue text-black"
   468→                    : "bg-gradient-to-br from-gray-800 to-gray-900 text-white",
   469→                  "shadow-lg"
   470→                )}
   471→              >
   472→                {message.role === 'user' ? (
   473→                  <User className="w-4 h-4" />
   474→                ) : (
   475→                  <Bot className="w-4 h-4" />
   476→                )}
   477→              </div>
   478→
   479→              {/* Message bubble */}
   480→              <div
   481→                className={cn(
   482→                  "max-w-[80%] md:max-w-[70%] px-5 py-3.5",
   483→                  message.role === 'user' 
   484→                    ? "message-user" 
   485→                    : "message-ai"
   486→                )}
   487→              >
   488→                <p className="text-sm leading-relaxed">{message.content}</p>
   489→                
   490→                {/* ✅ ACTION BUTTONS - Show only for AI messages with showButtons=true */}
   491→                {message.role === 'assistant' && message.showButtons && (
   492→                  <div 
   493→                    className="flex items-center gap-1 mt-3 pt-3 border-t border-white/10 animate-fadeIn"
   494→                  >
   495→                    {/* Timestamp */}
   496→                    <span className="text-xs text-muted-foreground mr-2">
   497→                      {message.timestamp.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
   498→                    </span>
   499→                    
   500→                    {/* 📋 COPY BUTTON - ORANGE THEME */}

   502→                      onClick={(e) => {
   503→                        e.preventDefault()
   504→                        handleCopy(message.content, message.id)

   507→                      title="Copy to clipboard"
   508→                      type="button"
   509→                    >
   510→                      {copiedId === message.id ? (
   511→                        <Check className="w-4 h-4 text-neon-cyan" />

   516→                    
   517→                    {/* 👍 LIKE BUTTON - ORANGE THEME */}
   518→                    <button
   519→                      onClick={() => {
   520→                        handleLike(message.id)
   521→                      }}
   522→                      className={`p-2 rounded-lg active:scale-95 transition-all cursor-pointer ${
   523→                        likedMessages.has(message.id)
   524→                          ? 'bg-neon-cyan/25'
   525→                          : 'hover:bg-neon-cyan/15'
   526→                      }`}

   528→                      type="button"

   535→                        }`} 
   536→                      />
   537→                    </button>
   538→                    
   539→                    {/* 👎 DISLIKE BUTTON - AMBER THEME */}
   540→                    <button

   542→                        handleDislike(message.id)
   543→                      }}
   544→                      className={`p-2 rounded-lg active:scale-95 transition-all cursor-pointer ${

   547→                          : 'hover:bg-red-500/15'
   548→                      }`}
   549→                      title="Bad response"
   550→                      type="button"

   562→                    <button

   566→                        }
   567→                      }}
   568→                      disabled={regeneratingId === message.id || isTyping}
   569→                      className={`p-2 rounded-lg active:scale-95 transition-all cursor-pointer disabled:opacity-40 disabled:cursor-not-allowed ${
   570→                        regeneratingId === message.id
   571→                          ? 'bg-neon-purple/20 animate-spin'
   572→                          : 'hover:bg-neon-purple/15'

   579→                          regeneratingId === message.id

   582→                        }`} 
   583→                      />
   584→                    </button>
