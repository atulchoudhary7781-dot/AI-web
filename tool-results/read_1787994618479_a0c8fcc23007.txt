   618→// Render markdown-like content
   619→function renderMessageContent(
   620→  content: string, 
   621→  onCopy: (text: string) => void, 
   622→  copiedCode: boolean
   623→): React.ReactNode {
   624→  const lines = content.split('\n')
   625→  const elements: React.ReactNode[] = []
   626→  let currentParagraph = ''
   627→  let inCodeBlock = false
   628→  let codeContent = ''
   629→
   630→  const flushParagraph = () => {
   631→    if (currentParagraph.trim()) {
   632→      elements.push(
   633→        <p key={`p-${elements.length}`} className="text-gray-200 my-2 whitespace-pre-wrap leading-relaxed">
   634→          {formatInlineMarkdown(currentParagraph.trim())}
   635→        </p>
   636→      )
   637→      currentParagraph = ''
   638→    }
   639→  }
   640→
   641→  lines.forEach((line, lineIndex) => {
   642→    // Code block handling
   643→    if (line.trim().startsWith('```')) {
   644→      if (inCodeBlock) {
   645→        elements.push(
   646→          <div key={`code-${elements.length}`} className="my-4 relative group/code">
   647→            {/* Code header */}
   648→            <div className="flex items-center justify-between bg-gray-900 rounded-t-lg px-4 py-2 border-b border-gray-700">
   649→              <span className="text-xs text-gray-400 font-mono">code</span>
   650→              <button
   651→                onClick={() => onCopy(codeContent)}
   652→                className="flex items-center gap-1.5 px-2.5 py-1 bg-gray-800 hover:bg-gray-700 rounded-md text-xs text-gray-300 transition-colors opacity-0 group-hover/code:opacity-100"
   653→              >
   654→                {copiedCode ? (
   655→                  <>
   656→                    <Check className="w-3 h-3 text-green-400" />
   657→                    Copied!
   658→                  </>
   659→                ) : (
   660→                  <>
   661→                    <Copy className="w-3 h-3" />
   662→                    Copy
   663→                  </>
   664→                )}
   665→              </button>
   666→            </div>
   667→            <pre className="bg-gray-900/80 rounded-b-lg p-4 overflow-x-auto text-sm border border-gray-800 border-t-0">
   668→              <code className="text-emerald-400 font-mono leading-relaxed">{codeContent}</code>
   669→            </pre>
   670→          </div>
   671→        )
   672→        codeContent = ''
   673→        inCodeBlock = false
   674→      } else {
   675→        flushParagraph()
   676→        inCodeBlock = true
   677→      }
   678→      return
   679→    }
   680→
   681→    if (inCodeBlock) {
   682→      codeContent += line + '\n'
   683→      return
   684→    }
   685→
   686→    // Headers
   687→    if (line.startsWith('# ')) {
   688→      flushParagraph()
   689→      elements.push(<h1 key={lineIndex} className="text-2xl font-bold text-white mt-6 mb-3 text-gradient">{line.slice(2)}</h1>)
   690→      return
   691→    }
   692→    if (line.startsWith('## ')) {
   693→      flushParagraph()
   694→      elements.push(<h2 key={lineIndex} className="text-xl font-semibold text-white mt-5 mb-2.5">{line.slice(3)}</h2>)
   695→      return
   696→    }
   697→    if (line.startsWith('### ')) {
   698→      flushParagraph()
   699→      elements.push(<h3 key={lineIndex} className="text-lg font-medium text-cyan-400 mt-4 mb-2">{line.slice(4)}</h3>)
   700→      return
   701→    }
   702→
   703→    // Lists
   704→    if (line.match(/^\s*[-*]\s+/)) {
   705→      flushParagraph()
   706→      elements.push(
   707→        <li key={lineIndex} className="ml-4 text-gray-200 list-disc mb-1 marker:text-cyan-500">
   708→          {formatInlineMarkdown(line.replace(/^\s*[-*]\s+/, ''))}
   709→        </li>
   710→      )
   711→      return
   712→    }
   713→    if (line.match(/^\d+\.\s+/)) {
   714→      flushParagraph()
   715→      elements.push(
   716→        <li key={lineIndex} className="ml-4 text-gray-200 list-decimal mb-1 marker:text-violet-400">
   717→          {formatInlineMarkdown(line.replace(/^\d+\.\s+/, ''))}
   718→        </li>
   719→      )
   720→      return
   721→    }
   722→
   723→    // Horizontal rule
   724→    if (line.trim() === '---') {
   725→      flushParagraph()
   726→      elements.push(<hr key={lineIndex} className="my-4 border-gray-700/50" />)
   727→      return
   728→    }
   729→
   730→    // Regular paragraph
   731→    if (line.trim()) {
   732→      currentParagraph += (currentParagraph ? ' ' : '') + line
   733→    } else {
   734→      flushParagraph()
   735→    }
   736→  })
   737→
   738→  flushParagraph()
   739→
   740→  return elements.length > 0 ? elements : content
   741→}

   743→// Format inline markdown
   744→function formatInlineMarkdown(text: string): React.ReactNode {
   745→  const parts: React.ReactNode[] = []
   746→  const regex = /\*\*(.*?)\*\*|`(.*?)`|\*(.*?)\*/g
   747→  let lastIndex = 0
   748→  let match
   749→  
   750→  while ((match = regex.exec(text)) !== null) {
   751→    if (match.index > lastIndex) {
   752→      parts.push(text.slice(lastIndex, match.index))
   753→    }
   754→    
   755→    if (match[1]) {
   756→      parts.push(<strong key={match.index} className="font-semibold text-white">{match[1]}</strong>)
   757→    } else if (match[2]) {
   758→      parts.push(<code key={match.index} className="px-2 py-0.5 bg-gray-800/80 rounded-md text-cyan-300 text-sm font-mono border border-gray-700/50">{match[2]}</code>)
   759→    } else if (match[3]) {
   760→      parts.push(<em key={match.index} className="italic text-gray-300">{match[3]}</em>)
   761→    }
   762→    
   763→    lastIndex = regex.lastIndex
   764→  }
   765→  
   766→  if (lastIndex < text.length) {
   767→    parts.push(text.slice(lastIndex))
   768→  }
   769→  
   770→  return parts.length > 0 ? parts : text
   771→}
