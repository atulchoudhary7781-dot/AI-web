    61→

    65→  // State

    78→      const userData = localStorage.getItem('user')
    79→      return userData ? JSON.parse(userData) : null
    80→    }
    81→    return null
    82→  }
    83→
    84→  const currentUser = getUser()
    85→
    86→  // Fetch stats on mount
    87→  useEffect(() => {

    90→      router.push('/')

    98→  const fetchStats = async () => {

   100→      const response = await fetch('/api/admin/stats', {
   101→        headers: {
   102→          'x-user-email': currentUser?.email || ''
   103→        }
   104→      })
   105→      
   106→      if (response.ok) {
   107→        const data = await response.json()
   108→        setStats(data.data)
   109→      }

   111→      console.error('Failed to fetch stats:', error)
   112→    }
   113→  }

   116→    try {
   117→      setLoading(true)
   118→      const params = new URLSearchParams({
   119→        page: currentPage.toString(),
   120→        limit: '20',
   121→        ...(searchQuery && { search: searchQuery }),
   122→        ...(planFilter && { plan: planFilter })

   124→
   125→      const response = await fetch(`/api/admin/users?${params}`, {
   126→        headers: {
   127→          'x-user-email': currentUser?.email || ''
   128→        }
   129→      })

   131→      if (response.ok) {
   132→        const data = await response.json()
   133→        setUsers(data.data.users)
   134→      }

   136→      console.error('Failed to fetch users:', error)
   137→    } finally {
   138→      setLoading(false)
   139→    }

   144→    if (activeTab === 'users') {

   154→      const response = await fetch('/api/admin/users', {
   155→        method: 'PUT',
   156→        headers: {
   157→          'Content-Type': 'application/json',
   158→          'x-user-email': currentUser?.email || ''
   159→        },
   160→        body: JSON.stringify({ userId, action, ...data })
   161→      })
   162→
   163→      if (response.ok) {
   164→        fetchUsers() // Refresh list
   165→        fetchStats() // Refresh stats
   166→      }

   168→      console.error('User action failed:', error)

   170→      setUserActionLoading(null)
   171→    }

   181→        method: 'DELETE',
   182→        headers: {
   183→          'x-user-email': currentUser?.email || ''
   184→        }
   185→      })

   187→      if (response.ok) {
   188→        fetchUsers()
   189→        fetchStats()

   207→      year: 'numeric',
   208→      month: 'short',
   209→      day: 'numeric'
