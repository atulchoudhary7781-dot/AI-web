    62→    62→// Neural Network Background Component
    63→    63→function NeuralNetworkBackground() {
    64→    64→  const canvasRef = useRef<HTMLCanvasElement>(null)
    65→    65→
    66→    66→  useEffect(() => {
    67→    67→    const canvas = canvasRef.current
    68→    68→    if (!canvas) return
    69→    69→
    70→    70→    const ctx = canvas.getContext('2d')
    71→    71→    if (!ctx) return
    72→    72→
    73→    73→    let animationFrameId: number
    74→    74→    let mouseX = 0
    75→    75→    let mouseY = 0
    76→    76→
    77→    77→    const resizeCanvas = () => {
    78→    78→      canvas.width = window.innerWidth
    79→    79→      canvas.height = window.innerHeight
    80→    80→    }
    81→    81→
    82→    82→    resizeCanvas()
    83→    83→    window.addEventListener('resize', resizeCanvas)
    84→    84→
    85→    85→    const handleMouseMove = (e: MouseEvent) => {
    86→    86→      mouseX = e.clientX
    87→    87→      mouseY = e.clientY
    88→    88→    }
    89→    89→    window.addEventListener('mousemove', handleMouseMove)
    90→    90→
    91→    91→    // Nodes for neural network
    92→    92→    interface Node {
    93→    93→      x: number
    94→    94→      y: number
    95→    95→      vx: number
    96→    96→      vy: number
    97→    97→      radius: number
    98→    98→      opacity: number
    99→    99→    }
   100→   100→
   101→   101→    const nodes: Node[] = []
   102→   102→    const nodeCount = Math.min(80, Math.floor((window.innerWidth * window.innerHeight) / 15000))
   103→   103→
   104→   104→    for (let i = 0; i < nodeCount; i++) {
   105→   105→      nodes.push({
   106→   106→        x: Math.random() * canvas.width,
   107→   107→        y: Math.random() * canvas.height,
   108→   108→        vx: (Math.random() - 0.5) * 0.5,
   109→   109→        vy: (Math.random() - 0.5) * 0.5,
   110→   110→        radius: Math.random() * 2 + 1,
   111→   111→        opacity: Math.random() * 0.5 + 0.3
   112→   112→      })
   113→   113→    }
   114→   114→
   115→   115→    const animate = () => {
   116→   116→      ctx.fillStyle = 'rgba(0, 0, 10, 0.1)'
   117→   117→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   118→   118→
   119→   119→      // Update and draw nodes
   120→   120→      nodes.forEach((node, i) => {
   121→   121→        // Mouse attraction
   122→   122→        const dx = mouseX - node.x
   123→   123→        const dy = mouseY - node.y
   124→   124→        const dist = Math.sqrt(dx * dx + dy * dy)
   125→   125→        
   126→   126→        if (dist < 200) {
   127→   127→          node.vx += dx * 0.00005
   128→   128→          node.vy += dy * 0.00005
   129→   129→        }
   130→   130→
   131→   131→        node.x += node.vx
   132→   132→        node.y += node.vy
   133→   133→
   134→   134→        // Bounce off edges
   135→   135→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   136→   136→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   137→   137→
   138→   138→        // Draw connections
   139→   139→        nodes.slice(i + 1).forEach(otherNode => {
   140→   140→          const distance = Math.sqrt(
   141→   141→            Math.pow(node.x - otherNode.x, 2) + Math.pow(node.y - otherNode.y, 2)
   142→   142→          )
   143→   143→          
   144→   144→          if (distance < 150) {
   145→   145→            const opacity = (1 - distance / 150) * 0.3
   146→   146→            ctx.beginPath()
   147→   147→            ctx.strokeStyle = `rgba(0, 245, 255, ${opacity})`
   148→   148→            ctx.lineWidth = 0.5
   149→   149→            ctx.moveTo(node.x, node.y)
   150→   150→            ctx.lineTo(otherNode.x, otherNode.y)
   151→   151→            ctx.stroke()
   152→   152→          }
   153→   153→        })
   154→   154→
   155→   155→        // Draw node
   156→   156→        ctx.beginPath()
   157→   157→        const gradient = ctx.createRadialGradient(
   158→   158→          node.x, node.y, 0,
   159→   159→          node.x, node.y, node.radius * 2
   160→   160→        )
   161→   161→        gradient.addColorStop(0, `rgba(0, 245, 255, ${node.opacity})`)
   162→   162→        gradient.addColorStop(1, 'rgba(124, 58, 237, 0)')
   163→   163→        ctx.fillStyle = gradient
   164→   164→        ctx.arc(node.x, node.y, node.radius * 2, 0, Math.PI * 2)
   165→   165→        ctx.fill()
   166→   166→      })
   167→   167→
   168→   168→      animationFrameId = requestAnimationFrame(animate)
   169→   169→    }
   170→   170→
   171→   171→    animate()
   172→   172→
   173→   173→    return () => {
   174→   174→      window.removeEventListener('resize', resizeCanvas)
   175→   175→      window.removeEventListener('mousemove', handleMouseMove)
   176→   176→      cancelAnimationFrame(animationFrameId)
   177→   177→    }
   178→   178→  }, [])
   179→   179→
   180→   180→  return (
   181→   181→    <canvas
   182→   182→      ref={canvasRef}
   183→   183→      className="fixed inset-0 pointer-events-none z-0"
   184→   184→      style={{ background: '#00000a' }}
   185→   185→    />
   186→   186→  )
   187→   187→}

   189→   189→// Glitch Text Component
   190→   190→function GlitchText({ text, className = '' }: { text: string; className?: string }) {
   191→   191→  const [glitchActive, setGlitchActive] = useState(false)
   192→   192→
   193→   193→  useEffect(() => {
   194→   194→    const interval = setInterval(() => {
   195→   195→      setGlitchActive(true)
   196→   196→      setTimeout(() => setGlitchActive(false), 200)
   197→   197→    }, 3000)
   198→   198→    return () => clearInterval(interval)
   199→   199→  }, [])
   200→   200→
   201→   201→  return (
   202→   202→    <span className={`relative inline-block ${className}`}>
   203→   203→      <span className={`relative z-10 ${glitchActive ? 'animate-pulse' : ''}`}>
   204→   204→        {text}
   205→   205→      </span>
   206→   206→      {glitchActive && (
   207→   207→        <>
   208→   208→          <span className="absolute top-0 left-0.5 text-[#ff00aa] opacity-80 clip-text-glitch-1" aria-hidden="true">
   209→   209→            {text}
   210→   210→          </span>
   211→   211→          <span className="absolute top-0 -left-0.5 text-[#00f5ff] opacity-80 clip-text-glitch-2" aria-hidden="true">
   212→   212→            {text}
   213→   213→          </span>
   214→   214→        </>
   215→   215→      )}
   216→   216→    </span>
   217→   217→  )
   218→   218→}

   220→   220→// Animated Counter Component
   221→   221→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
   222→   222→  const [count, setCount] = useState(0)
   223→   223→  const [isVisible, setIsVisible] = useState(false)
   224→   224→  const ref = useRef<HTMLDivElement>(null)
   225→   225→
   226→   226→  useEffect(() => {
   227→   227→    const observer = new IntersectionObserver(
   228→   228→      ([entry]) => {
   229→   229→        if (entry.isIntersecting && !isVisible) {
   230→   230→          setIsVisible(true)
   231→   231→        }
   232→   232→      },
   233→   233→      { threshold: 0.1 }
   234→   234→    )
   235→   235→
   236→   236→    if (ref.current) observer.observe(ref.current)
   237→   237→    return () => observer.disconnect()
   238→   238→  }, [isVisible])
   239→   239→
   240→   240→  useEffect(() => {
   241→   241→    if (!isVisible) return
   242→   242→    
   243→   243→    let start = 0
   244→   244→    const increment = target / (duration / 16)
   245→   245→    
   246→   246→    const timer = setInterval(() => {
   247→   247→      start += increment
   248→   248→      if (start >= target) {
   249→   249→        setCount(target)
   250→   250→        clearInterval(timer)
   251→   251→      } else {
   252→   252→        setCount(Math.floor(start))
   253→   253→      }
   254→   254→    }, 16)
   255→   255→
   256→   256→    return () => clearInterval(timer)
   257→   257→  }, [isVisible, target, duration])
   258→   258→
   259→   259→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
   260→   260→}
