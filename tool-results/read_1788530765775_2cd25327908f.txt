   320→// ==================== MAIN COMPONENT ====================
   321→export function ChatInterface() {
   322→  const { t } = useI18n()
   323→  
   324→  // State
   325→  const [messages, setMessages] = useState<Message[]>([
   326→    {
   327→      id: '1',
   328→      role: 'assistant',
   329→      content: "Greetings, human! I am NEXUS AI, your advanced neural companion. How may I assist you today?",
   330→      timestamp: new Date(),
   331→      showButtons: true,
   332→    }
   333→  ])
   334→  const [input, setInput] = useState('')
   335→  const [isTyping, setIsTyping] = useState(false)
   336→  const [copiedId, setCopiedId] = useState<string | null>(null)
   337→  const [isExpanded, setIsExpanded] = useState(false)
   338→  const [showHistory, setShowHistory] = useState(false)
   339→  const [currentConversation, setCurrentConversation] = useState<ChatConversation | null>(null)
   340→  
   341→  // AI Model State - Default to Llama 3.1 (FREE)
   342→  const [selectedModel, setSelectedModel] = useState<AIModel>(DEFAULT_MODEL!)
   343→  const [showModelSelector, setShowModelSelector] = useState(false)
   344→  
   345→  // Reaction states
   346→  const [likedMessages, setLikedMessages] = useState<Set<string>>(new Set())
   347→  const [dislikedMessages, setDislikedMessages] = useState<Set<string>>(new Set())
   348→  const [regeneratingId, setRegeneratingId] = useState<string | null>(null)
   349→  
   350→  // Refs
   351→  const messagesEndRef = useRef<HTMLDivElement>(null)
   352→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   353→
   354→  // Auto-scroll to bottom
   355→  useEffect(() => {
   356→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   357→  }, [messages])
   358→
   359→  // Handle voice input transcript
   360→  const handleVoiceTranscript = useCallback((transcript: string) => {
   361→    setInput(prev => prev ? `${prev} ${transcript}` : transcript)
   362→    textareaRef.current?.focus()
   363→  }, [])
   364→
   365→  // Handle send message
   366→  const handleSend = useCallback(async () => {
   367→    if (!input.trim()) return
   368→
   369→    const userMessage: Message = {
   370→      id: 'user-' + Date.now(),
   371→      role: 'user',
   372→      content: input.trim(),
   373→      timestamp: new Date(),
   374→    }
   375→
   376→    setMessages((prev) => [...prev, userMessage])
   377→    setInput('')
   378→    setIsTyping(true)
   379→
   380→    // Simulate AI response
   381→    setTimeout(() => {
   382→      const aiMessageId = 'ai-' + Date.now()
   383→      const aiMessage: Message = {
   384→        id: aiMessageId,
   385→        role: 'assistant',
   386→        content: `🤖 **${selectedModel.name}** (${selectedModel.provider})\n\nI've analyzed your query using ${selectedModel.name}. How can I help you further?`,
   387→        timestamp: new Date(),
   388→        showButtons: false,
   389→      }
   390→      
   391→      setMessages((prev) => [...prev, aiMessage])
   392→      setIsTyping(false)
   393→      
   394→      // Show buttons after delay
   395→      setTimeout(() => {
   396→        setMessages(prev => prev.map(msg => 
   397→          msg.id === aiMessageId ? { ...msg, showButtons: true } : msg
   398→        ))
   399→      }, 800)
   400→    }, 1500 + Math.random() * 1000)
   401→  }, [input, selectedModel])
   402→
   403→  // Handle copy message
   404→  const handleCopy = useCallback(async (content: string, id: string) => {
   405→    try {
   406→      await navigator.clipboard.writeText(content)
   407→      setCopiedId(id)
   408→      setTimeout(() => setCopiedId(null), 2000)
   409→    } catch (err) {
   410→      console.error('Failed to copy:', err)
   411→    }
   412→  }, [])
   413→
   414→  // Handle reactions
   415→  const handleLike = useCallback((id: string) => {
   416→    setLikedMessages(prev => {
   417→      const newSet = new Set(prev)
   418→      if (newSet.has(id)) newSet.delete(id)
   419→      else newSet.add(id)
   420→      return newSet
   421→    })
   422→    setDislikedMessages(prev => {
   423→      const newSet = new Set(prev)
   424→      newSet.delete(id)
   425→      return newSet
   426→    })
   427→  }, [])
   428→
   429→  const handleDislike = useCallback((id: string) => {
   430→    setDislikedMessages(prev => {
   431→      const newSet = new Set(prev)
   432→      if (newSet.has(id)) newSet.delete(id)
   433→      else newSet.add(id)
   434→      return newSet
   435→    })
   436→    setLikedMessages(prev => {
   437→      const newSet = new Set(prev)
   438→      newSet.delete(id)
   439→      return newSet
   440→    })
   441→  }, [])
   442→
   443→  // Handle regenerate response
   444→  const handleRegenerate = useCallback((id: string) => {
   445→    setRegeneratingId(id)
   446→    
   447→    setTimeout(() => {
   448→      const regeneratedContent = `🔄 **Regenerated Response (${selectedModel.name}**)\n\nHere's my fresh perspective based on ${selectedModel.provider}'s capabilities.\n\nWould you like me to explore anything specific?`
   449→      
   450→      setMessages(prev => prev.map(msg => 
   451→        msg.id === id ? { ...msg, content: regeneratedContent, showButtons: true } : msg
   452→      ))
   453→      setRegeneratingId(null)
   454→    }, 1500 + Math.random() * 1000)
   455→  }, [selectedModel])
   456→
   457→  // Handle keyboard input
   458→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   459→    if (e.key === 'Enter' && !e.shiftKey) {
   460→      e.preventDefault()
   461→      handleSend()
   462→    }
   463→  }, [handleSend])
   464→
   465→  // Clear chat
   466→  const handleClearChat = useCallback(() => {
   467→    if (window.confirm(t('chat.clearConfirm'))) {
   468→      setMessages([])
   469→      setLikedMessages(new Set())
   470→      setDislikedMessages(new Set())
   471→    }
   472→  }, [t])
   473→
   474→  // Handle model selection
   475→  const handleModelSelect = useCallback((model: AIModel) => {
   476→    if (model.isLocked) {
   477→      alert('🔒 Subscription Required!\n\nThis model is only available with a premium subscription.\nComing soon!')
   478→      return
   479→    }
   480→    setSelectedModel(model)
   481→    setShowModelSelector(false)
   482→  }, [])
   483→
   484→  // Memoized values
   485→  const sortedModels = useMemo(() => [...AI_MODELS].sort((a, b) => {
   486→    if (a.isLocked === b.isLocked) return 0
   487→    return a.isLocked ? 1 : -1
   488→  }), [])
   489→
   490→  return (
   491→    <div className="flex flex-col h-[100dvh] bg-dark-bg text-foreground font-sans antialiased overflow-hidden">
   492→      {/* ==================== HEADER - MOBILE OPTIMIZED ==================== */}
   493→      <header className="flex-shrink-0 flex items-center justify-between px-3 sm:px-6 py-2 sm:py-3 border-b border-white/10 bg-dark-surface/95 backdrop-blur-md z-20 safe-area-top">
   494→        {/* Left: Logo & Title */}
   495→        <div className="flex items-center gap-2 sm:gap-4 min-w-0">
   496→          <h1 className="text-base sm:text-xl font-bold bg-gradient-to-r from-neon-cyan via-neon-purple to-electric-blue bg-clip-text text-transparent truncate">
   497→            NEXUS AI
   498→          </h1>
   499→          <Badge variant="cyberpunk" className="hidden md:inline-flex text-[10px]">
   500→            Neural v4.0
   501→          </Badge>
   502→        </div>
   503→        
   504→        {/* Right: Action Buttons - Touch friendly (44px+ tap target) */}
   505→        <div className="flex items-center gap-1 sm:gap-2">
   506→          {/* Clear chat - Hidden on small mobile */}
   507→          <Button
   508→            variant="ghost"
   509→            size="icon"
   510→            onClick={handleClearChat}
   511→            className="hidden sm:flex text-muted-foreground hover:text-red-400 h-9 w-9"
   512→          >
   513→            <Trash2 className="w-4 h-4" />
   514→          </Button>
   515→          
   516→          {/* History */}
   517→          <Button
   518→            variant="ghost"
   519→            size="icon"
   520→            onClick={() => setShowHistory(!showHistory)}
   521→            className="text-muted-foreground hover:text-neon-cyan h-9 w-9 sm:h-10 sm:w-10"
   522→          >
   523→            <History className="w-4 h-4 sm:w-5 sm:h-5" />
   524→          </Button>
   525→
   526→          {/* Expand/Collapse - Hidden on mobile */}
   527→          <Button
   528→            variant="ghost"
   529→            size="icon"
   530→            onClick={() => setIsExpanded(!isExpanded)}
   531→            className="hidden sm:flex text-muted-foreground hover:text-neon-cyan h-9 w-9"
   532→          >
   533→            {isExpanded ? <Minimize2 className="w-4 h-4" /> : <Maximize2 className="w-4 h-4" />}
   534→          </Button>
   535→        </div>
   536→      </header>
