    61→// Neural Network Background Component
    62→function NeuralNetworkBackground() {
    63→  const canvasRef = useRef<HTMLCanvasElement>(null)
    64→
    65→  useEffect(() => {
    66→    const canvas = canvasRef.current
    67→    if (!canvas) return
    68→
    69→    const ctx = canvas.getContext('2d')
    70→    if (!ctx) return
    71→
    72→    let animationFrameId: number
    73→    let mouseX = 0
    74→    let mouseY = 0
    75→
    76→    const resizeCanvas = () => {
    77→      canvas.width = window.innerWidth
    78→      canvas.height = window.innerHeight
    79→    }
    80→
    81→    resizeCanvas()
    82→    window.addEventListener('resize', resizeCanvas)
    83→
    84→    const handleMouseMove = (e: MouseEvent) => {
    85→      mouseX = e.clientX
    86→      mouseY = e.clientY
    87→    }
    88→    window.addEventListener('mousemove', handleMouseMove)
    89→
    90→    // Nodes for neural network
    91→    interface Node {
    92→      x: number
    93→      y: number
    94→      vx: number
    95→      vy: number
    96→      radius: number
    97→      opacity: number
    98→    }
    99→
   100→    const nodes: Node[] = []
   101→    const nodeCount = Math.min(80, Math.floor((window.innerWidth * window.innerHeight) / 15000))
   102→
   103→    for (let i = 0; i < nodeCount; i++) {
   104→      nodes.push({
   105→        x: Math.random() * canvas.width,
   106→        y: Math.random() * canvas.height,
   107→        vx: (Math.random() - 0.5) * 0.5,
   108→        vy: (Math.random() - 0.5) * 0.5,
   109→        radius: Math.random() * 2 + 1,
   110→        opacity: Math.random() * 0.5 + 0.3
   111→      })
   112→    }
   113→
   114→    const animate = () => {
   115→      ctx.fillStyle = 'rgba(0, 0, 10, 0.1)'
   116→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   117→
   118→      // Update and draw nodes
   119→      nodes.forEach((node, i) => {
   120→        // Mouse attraction
   121→        const dx = mouseX - node.x
   122→        const dy = mouseY - node.y
   123→        const dist = Math.sqrt(dx * dx + dy * dy)
   124→        
   125→        if (dist < 200) {
   126→          node.vx += dx * 0.00005
   127→          node.vy += dy * 0.00005
   128→        }
   129→
   130→        node.x += node.vx
   131→        node.y += node.vy
   132→
   133→        // Bounce off edges
   134→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   135→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   136→
   137→        // Draw connections
   138→        nodes.slice(i + 1).forEach(otherNode => {
   139→          const distance = Math.sqrt(
   140→            Math.pow(node.x - otherNode.x, 2) + Math.pow(node.y - otherNode.y, 2)
   141→          )
   142→          
   143→          if (distance < 150) {
   144→            const opacity = (1 - distance / 150) * 0.3
   145→            ctx.beginPath()
   146→            ctx.strokeStyle = `rgba(0, 245, 255, ${opacity})`
   147→            ctx.lineWidth = 0.5
   148→            ctx.moveTo(node.x, node.y)
   149→            ctx.lineTo(otherNode.x, otherNode.y)
   150→            ctx.stroke()
   151→          }
   152→        })
   153→
   154→        // Draw node
   155→        ctx.beginPath()
   156→        const gradient = ctx.createRadialGradient(
   157→          node.x, node.y, 0,
   158→          node.x, node.y, node.radius * 2
   159→        )
   160→        gradient.addColorStop(0, `rgba(0, 245, 255, ${node.opacity})`)
   161→        gradient.addColorStop(1, 'rgba(124, 58, 237, 0)')
   162→        ctx.fillStyle = gradient
   163→        ctx.arc(node.x, node.y, node.radius * 2, 0, Math.PI * 2)
   164→        ctx.fill()
   165→      })
   166→
   167→      animationFrameId = requestAnimationFrame(animate)
   168→    }
   169→
   170→    animate()
   171→
   172→    return () => {
   173→      window.removeEventListener('resize', resizeCanvas)
   174→      window.removeEventListener('mousemove', handleMouseMove)
   175→      cancelAnimationFrame(animationFrameId)
   176→    }
   177→  }, [])
   178→
   179→  return (
   180→    <canvas
   181→      ref={canvasRef}
   182→      className="fixed inset-0 pointer-events-none z-0"
   183→      style={{ background: '#00000a' }}
   184→    />
   185→  )
   186→}

   188→// Glitch Text Component
   189→function GlitchText({ text, className = '' }: { text: string; className?: string }) {
   190→  const [glitchActive, setGlitchActive] = useState(false)
   191→
   192→  useEffect(() => {
   193→    const interval = setInterval(() => {
   194→      setGlitchActive(true)
   195→      setTimeout(() => setGlitchActive(false), 200)
   196→    }, 3000)
   197→    return () => clearInterval(interval)
   198→  }, [])
   199→
   200→  return (
   201→    <span className={`relative inline-block ${className}`}>
   202→      <span className={`relative z-10 ${glitchActive ? 'animate-pulse' : ''}`}>
   203→        {text}
   204→      </span>
   205→      {glitchActive && (
   206→        <>
   207→          <span className="absolute top-0 left-0.5 text-[#ff00aa] opacity-80 clip-text-glitch-1" aria-hidden="true">
   208→            {text}
   209→          </span>
   210→          <span className="absolute top-0 -left-0.5 text-[#00f5ff] opacity-80 clip-text-glitch-2" aria-hidden="true">
   211→            {text}
   212→          </span>
   213→        </>
   214→      )}
   215→    </span>

   219→// Animated Counter Component
   220→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
   221→  const [count, setCount] = useState(0)
   222→  const [isVisible, setIsVisible] = useState(false)
   223→  const ref = useRef<HTMLDivElement>(null)
   224→
   225→  useEffect(() => {
   226→    const observer = new IntersectionObserver(
   227→      ([entry]) => {
   228→        if (entry.isIntersecting && !isVisible) {
   229→          setIsVisible(true)
   230→        }
   231→      },
   232→      { threshold: 0.1 }
   233→    )
   234→
   235→    if (ref.current) observer.observe(ref.current)
   236→    return () => observer.disconnect()
   237→  }, [isVisible])
   238→
   239→  useEffect(() => {
   240→    if (!isVisible) return
   241→    
   242→    let start = 0
   243→    const increment = target / (duration / 16)
   244→    
   245→    const timer = setInterval(() => {
   246→      start += increment
   247→      if (start >= target) {
   248→        setCount(target)
   249→        clearInterval(timer)
   250→      } else {
   251→        setCount(Math.floor(start))
   252→      }
   253→    }, 16)
   254→
   255→    return () => clearInterval(timer)
   256→  }, [isVisible, target, duration])
   257→
   258→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
   259→}
