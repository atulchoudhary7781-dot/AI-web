    25→// ============ ANIMATED BACKGROUND COMPONENT ============
    26→function ParticleBackground() {
    27→  const [mousePos, setMousePos] = useState({ x: 0, y: 0 })
    28→  const [particles, setParticles] = useState<Array<{ id: number; x: number; y: number; size: number; duration: number; delay: number }>>([])
    29→
    30→  useEffect(() => {
    31→    const newParticles = Array.from({ length: 50 }, (_, i) => ({
    32→      id: i,
    33→      x: Math.random() * 100,
    34→      y: Math.random() * 100,
    35→      size: Math.random() * 4 + 1,
    36→      duration: Math.random() * 20 + 10,
    37→      delay: Math.random() * 5
    38→    }))
    39→    setParticles(newParticles)
    40→  }, [])
    41→
    42→  useEffect(() => {
    43→    const handleMouseMove = (e: MouseEvent) => {
    44→      setMousePos({ x: e.clientX, y: e.clientY })
    45→    }
    46→    window.addEventListener('mousemove', handleMouseMove)
    47→    return () => window.removeEventListener('mousemove', handleMouseMove)
    48→  }, [])
    49→
    50→  return (
    51→    <div className="fixed inset-0 overflow-hidden pointer-events-none z-0">
    52→      {/* Gradient orbs that follow mouse */}
    53→      <motion.div
    54→        className="absolute w-[600px] h-[600px] rounded-full opacity-20 blur-[120px]"
    55→        style={{
    56→          background: 'radial-gradient(circle, rgba(139, 92, 246, 0.4) 0%, transparent 70%)',
    57→          left: mousePos.x - 300,
    58→          top: mousePos.y - 300,
    59→        }}
    60→        animate={{
    61→          x: 0,
    62→          y: 0,
    63→        }}
    64→        transition={{ type: 'spring', stiffness: 50, damping: 30 }}
    65→      />
    66→      <motion.div
    67→        className="absolute w-[400px] h-[400px] rounded-full opacity-15 blur-[100px]"
    68→        style={{
    69→          background: 'radial-gradient(circle, rgba(59, 130, 246, 0.5) 0%, transparent 70%)',
    70→          right: -mousePos.x * 0.2,
    71→          top: -mousePos.y * 0.2,
    72→        }}
    73→        animate={{
    74→          x: 0,
    75→          y: 0,
    76→        }}
    77→        transition={{ type: 'spring', stiffness: 30, damping: 25 }}
    78→      />
    79→      
    80→      {/* Floating particles */}
    81→      {particles.map((particle) => (
    82→        <motion.div
    83→          key={particle.id}
    84→          className="absolute rounded-full bg-white/30"
    85→          style={{
    86→            width: particle.size,
    87→            height: particle.size,
    88→            left: `${particle.x}%`,
    89→            top: `${particle.y}%`,
    90→          }}
    91→          animate={{
    92→            y: [0, -100, 0],
    93→            opacity: [0.3, 0.8, 0.3],
    94→            scale: [1, 1.5, 1],
    95→          }}
    96→          transition={{
    97→            duration: particle.duration,
    98→            repeat: Infinity,
    99→            delay: particle.delay,
   100→            ease: 'easeInOut'
   101→          }}
   102→        />
   103→      ))}
   104→      
   105→      {/* Grid pattern overlay */}
   106→      <div 
   107→        className="absolute inset-0 opacity-[0.03]"
   108→        style={{
   109→          backgroundImage: `linear-gradient(rgba(255,255,255,.1) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,.1) 1px, transparent 1px)`,
   110→          backgroundSize: '50px 50px'
   111→        }}
   112→      />
   113→    </div>
   114→  )
   115→}

   118→function Navigation() {

   125→      setScrolled(window.scrollY > 50)

   128→      for (const section of sections.reverse()) {
