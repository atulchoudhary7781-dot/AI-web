     1→     1→'use client'
     2→     2→
     3→     3→import { useState, useRef, useEffect } from 'react'
     4→     4→import { 
     5→     5→  Search, Image, Mic, Code2, FileText, Languages, 
     6→     6→  X, Send, Loader2, Copy, Check, Download,
     7→     7→  Play, Square, Volume2, VolumeX, Upload,
     8→     8→  ExternalLink, ChevronDown, ChevronUp, Sparkles
     9→     9→} from 'lucide-react'
    10→    10→import { Button } from '@/components/ui/button'
    11→    11→
    12→    12→interface ToolsPanelProps {
    13→    13→  isOpen: boolean
    14→    14→  onClose: () => void
    15→    15→  onInsertToChat: (text: string) => void
    16→    16→}
    17→    17→
    18→    18→type ToolType = 'web-search' | 'image-gen' | 'voice-chat' | 'code-executor' | 'file-analyzer' | 'translator' | null
    19→    19→
    20→    20→export default function ToolsPanel({ isOpen, onClose, onInsertToChat }: ToolsPanelProps) {
    21→    21→  const [activeTool, setActiveTool] = useState<ToolType>(null)
    22→    22→  
    23→    23→  // Tool states
    24→    24→  const [searchQuery, setSearchQuery] = useState('')
    25→    25→  const [searchResults, setSearchResults] = useState<any[]>([])
    26→    26→  const [isSearching, setIsSearching] = useState(false)
    27→    27→  
    28→    28→  const [imagePrompt, setImagePrompt] = useState('')
    29→    29→  const [imageSize, setImageSize] = useState('1024x1024')
    30→    30→  const [generatedImage, setGeneratedImage] = useState<string | null>(null)
    31→    31→  const [isGeneratingImage, setIsGeneratingImage] = useState(false)
    32→    32→  
    33→    33→  const [voiceText, setVoiceText] = useState('')
    34→    34→  const [isRecording, setIsRecording] = useState(false)
    35→    35→  const [isSpeaking, setIsSpeaking] = useState(false)
    36→    36→  const [selectedVoice, setSelectedVoice] = useState('alloy')
    37→    37→  
    38→    38→  const [code, setCode] = useState('')
    39→    39→  const [codeLanguage, setCodeLanguage] = useState('javascript')
    40→    40→  const [codeOutput, setCodeOutput] = useState<string>('')
    41→    41→  const [codeError, setCodeError] = useState<string>('')
    42→    42→  const [isRunningCode, setIsRunningCode] = useState(false)
    43→    43→  
    44→    44→  const [selectedFile, setSelectedFile] = useState<File | null>(null)
    45→    45→  const [analysisResult, setAnalysisResult] = useState<any>(null)
    46→    46→  const [isAnalyzing, setIsAnalyzing] = useState(false)
    47→    47→  
    48→    48→  const [translateText, setTranslateText] = useState('')
    49→    49→  const [sourceLang, setSourceLang] = useState('auto')
    50→    50→  const [targetLang, setTargetLang] = useState('hi')
    51→    51→  const [translatedText, setTranslatedText] = useState<string>('')
    52→    52→  const [isTranslating, setIsTranslating] = useState(false)
    53→    53→
    54→    54→  const fileInputRef = useRef<HTMLInputElement>(null)
    55→    55→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
    56→    56→
    57→    57→  // Reset states when tool changes
    58→    58→  useEffect(() => {
    59→    59→    setActiveTool(null)
    60→    60→    setSearchResults([])
    61→    61→    setGeneratedImage(null)
    62→    62→    setCodeOutput('')
    63→    63→    setCodeError('')
    64→    64→    setAnalysisResult(null)
    65→    65→    setTranslatedText('')
    66→    66→  }, [])
    67→    67→
    68→    68→  if (!isOpen) return null
    69→    69→
    70→    70→  // Tool handlers
    71→    71→  const handleWebSearch = async () => {
    72→    72→    if (!searchQuery.trim()) return
    73→    73→    
    74→    74→    setIsSearching(true)
    75→    75→    try {
    76→    76→      const response = await fetch('/api/tools/web-search', {
    77→    77→        method: 'POST',
    78→    78→        headers: { 'Content-Type': 'application/json' },
    79→    79→        body: JSON.stringify({ query: searchQuery, numResults: 5 })
    80→    80→      })
    81→    81→      const data = await response.json()
    82→    82→      
    83→    83→      if (data.success) {
    84→    84→        setSearchResults(data.results)
    85→    85→      }
    86→    86→    } catch (error) {
    87→    87→      console.error('Search error:', error)
    88→    88→    }
    89→    89→    setIsSearching(false)
    90→    90→  }
    91→    91→
    92→    92→  const handleGenerateImage = async () => {
    93→    93→    if (!imagePrompt.trim()) return
    94→    94→    
    95→    95→    setIsGeneratingImage(true)
    96→    96→    try {
    97→    97→      const response = await fetch('/api/tools/generate-image', {
    98→    98→        method: 'POST',
    99→    99→        headers: { 'Content-Type': 'application/json' },
   100→   100→        body: JSON.stringify({ prompt: imagePrompt, size: imageSize })
   101→   101→      })
   102→   102→      const data = await response.json()
   103→   103→      
   104→   104→      if (data.success && data.image) {
   105→   105→        setGeneratedImage(data.image.base64 || data.image.url)
   106→   106→      }
   107→   107→    } catch (error) {
   108→   108→      console.error('Image generation error:', error)
   109→   109→    }
   110→   110→    setIsGeneratingImage(false)
   111→   111→  }
   112→   112→
   113→   113→  const toggleRecording = async () => {
   114→   114→    if (!isRecording) {
   115→   115→      try {
   116→   116→        const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   117→   117→        mediaRecorderRef.current = new MediaRecorder(stream)
   118→   118→        const chunks: Blob[] = []
   119→   119→        
   120→   120→        mediaRecorderRef.current.ondataavailable = (e) => chunks.push(e.data)
   121→   121→        mediaRecorderRef.current.onstop = async () => {
   122→   122→          const blob = new Blob(chunks, { type: 'audio/webm' })
   123→   123→          
   124→   124→          // Send to STT API
   125→   125→          try {
   126→   126→            const formData = new FormData()
   127→   127→            formData.append('audioData', blob)
   128→   128→            
   129→   129→            const response = await fetch('/api/tools/voice-chat', {
   130→   130→              method: 'POST',
   131→   131→              body: JSON.stringify({ action: 'stt', audioData: await blob.arrayBuffer() }),
   132→   132→              headers: { 'Content-Type': 'application/json' }
   133→   133→            })
   134→   134→            const data = await response.json()
   135→   135→            
   136→   136→            if (data.success && data.text) {
   137→   137→              setVoiceText(prev => prev + (prev ? ' ' : '') + data.text.replace(/\[.*?\]/g, '').trim())
   138→   138→            }
   139→   139→          } catch (error) {
   140→   140→            console.error('STT error:', error)
   141→   141→          }
   142→   142→          
   143→   143→          stream.getTracks().forEach(track => track.stop())
   144→   144→        }
   145→   145→        
   146→   146→        mediaRecorderRef.current.start()
   147→   147→        setIsRecording(true)
   148→   148→      } catch (error) {
   149→   149→        console.error('Microphone access denied:', error)
   150→   150→        alert('Microphone access is required for voice input.')
   151→   151→      }
   152→   152→    } else {
   153→   153→      mediaRecorderRef.current?.stop()
   154→   154→      setIsRecording(false)
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  const handleSpeak = async () => {
   159→   159→    if (!voiceText.trim() || isSpeaking) return
   160→   160→    
   161→   161→    setIsSpeaking(true)
   162→   162→    try {
   163→   163→      const response = await fetch('/api/tools/voice-chat', {
   164→   164→        method: 'POST',
   165→   165→        headers: { 'Content-Type': 'application/json' },
   166→   166→        body: JSON.stringify({ action: 'tts', text: voiceText, voice: selectedVoice })
   167→   167→      })
   168→   168→      const data = await response.json()
   169→   169→      
   170→   170→      if (data.success && data.audioUrl) {
   171→   171→        // Play audio
   172→   172→        const audio = new Audio(data.audioUrl)
   173→   173→        audio.onended = () => setIsSpeaking(false)
   174→   174→        audio.play()
   175→   175→      } else {
   176→   176→        setTimeout(() => setIsSpeaking(false), 2000)
   177→   177→      }
   178→   178→    } catch (error) {
   179→   179→      console.error('TTS error:', error)
   180→   180→      setIsSpeaking(false)
   181→   181→    }
   182→   182→  }
   183→   183→
   184→   184→  const handleRunCode = async () => {
   185→   185→    if (!code.trim()) return
   186→   186→    
   187→   187→    setIsRunningCode(true)
   188→   188→    setCodeOutput('')
   189→   189→    setCodeError('')
   190→   190→    
   191→   191→    try {
   192→   192→      const response = await fetch('/api/tools/run-code', {
   193→   193→        method: 'POST',
   194→   194→        headers: { 'Content-Type': 'application/json' },
   195→   195→        body: JSON.stringify({ code, language: codeLanguage })
   196→   196→      })
   197→   197→      const data = await response.json()
   198→   198→      
   199→   199→      if (data.success) {
   200→   200→        setCodeOutput(data.output || '')
   201→   201→        setCodeError(data.error || '')
   202→   202→      } else {
   203→   203→        setCodeError(data.error || 'Execution failed')
   204→   204→      }
   205→   205→    } catch (error) {
   206→   206→      setCodeError('Failed to execute code')
   207→   207→    }
   208→   208→    
   209→   209→    setIsRunningCode(false)
   210→   210→  }
   211→   211→
   212→   212→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   213→   213→    const file = e.target.files?.[0]
   214→   214→    if (file) setSelectedFile(file)
   215→   215→  }
   216→   216→
   217→   217→  const handleAnalyzeFile = async () => {
   218→   218→    if (!selectedFile) return
   219→   219→    
   220→   220→    setIsAnalyzing(true)
   221→   221→    try {
   222→   222→      const formData = new FormData()
   223→   223→      formData.append('file', selectedFile)
   224→   224→      formData.append('analysisType', 'summary')
   225→   225→      
   226→   226→      const response = await fetch('/api/tools/analyze-file', {
   227→   227→        method: 'POST',
   228→   228→        body: formData
   229→   229→      })
   230→   230→      const data = await response.json()
   231→   231→      
   232→   232→      if (data.success) {
   233→   233→        setAnalysisResult(data.analysis)
   234→   234→      }
   235→   235→    } catch (error) {
   236→   236→      console.error('Analysis error:', error)
   237→   237→    }
   238→   238→    setIsAnalyzing(false)
   239→   239→  }
   240→   240→
   241→   241→  const handleTranslate = async () => {
   242→   242→    if (!translateText.trim()) return
   243→   243→    
   244→   244→    setIsTranslating(true)
   245→   245→    try {
   246→   246→      const response = await fetch('/api/tools/translate', {
   247→   247→        method: 'POST',
   248→   248→        headers: { 'Content-Type': 'application/json' },
   249→   249→        body: JSON.stringify({ text: translateText, sourceLang, targetLang })
   250→   250→      })
   251→   251→      const data = await response.json()
   252→   252→      
   253→   253→      if (data.success) {
   254→   254→        setTranslatedText(data.translatedText)
   255→   255→      }
   256→   256→    } catch (error) {
   257→   257→      console.error('Translation error:', error)
   258→   258→    }
   259→   259→    setIsTranslating(false)
   260→   260→  }
   261→   261→
   262→   262→  const tools = [
   263→   263→    { id: 'web-search', icon: Search, label: 'Web Search', color: 'from-blue-500 to-cyan-500', description: 'Search the internet' },
   264→   264→    { id: 'image-gen', icon: Image, label: 'AI Images', color: 'from-purple-500 to-pink-500', description: 'Generate images' },
   265→   265→    { id: 'voice-chat', icon: Mic, label: 'Voice Chat', color: 'from-green-500 to-emerald-500', description: 'Speak & listen' },
   266→   266→    { id: 'code-executor', icon: Code2, label: 'Code Runner', color: 'from-orange-500 to-yellow-500', description: 'Execute code' },
   267→   267→    { id: 'file-analyzer', icon: FileText, label: 'File Analyzer', color: 'from-red-500 to-rose-500', description: 'Analyze files' },
   268→   268→    { id: 'translator', icon: Languages, label: 'Translator', color: 'from-indigo-500 to-violet-500', description: 'Translate text' }
   269→   269→  ]
   270→   270→
   271→   271→  return (
   272→   272→    <div className="fixed inset-0 z-[100] bg-black/60 backdrop-blur-sm flex items-end sm:items-center justify-center p-0 sm:p-4">
   273→   273→      <div className="bg-gray-900 border border-gray-800 rounded-t-2xl sm:rounded-2xl w-full sm:max-w-4xl max-h-[90vh] overflow-hidden shadow-2xl flex flex-col">
   274→   274→        {/* Header */}
   275→   275→        <div className="flex items-center justify-between px-4 py-3 border-b border-gray-800 bg-gray-900/95">
   276→   276→          <div className="flex items-center gap-3">
   277→   277→            <Sparkles className="w-5 h-5 text-cyan-400" />
   278→   278→            <h2 className="text-lg font-semibold text-white">AI Tools</h2>
   279→   279→          </div>
   280→   280→          <button onClick={onClose} className="p-1.5 rounded-lg hover:bg-gray-800 transition-colors">
   281→   281→            <X className="w-5 h-5 text-gray-400" />
   282→   282→          </button>
   283→   283→        </div>
   284→   284→
   285→   285→        {!activeTool ? (
   286→   286→          /* Tool Selection Grid */
   287→   287→          <div className="p-4 overflow-y-auto">
   288→   288→            <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
   289→   289→              {tools.map((tool) => (
   290→   290→                <button
   291→   291→                  key={tool.id}
   292→   292→                  onClick={() => setActiveTool(tool.id as ToolType)}
   293→   293→                  className="group p-4 rounded-xl bg-gray-800/50 hover:bg-gray-800 border border-gray-700/50 hover:border-cyan-500/30 transition-all duration-200 text-left"
   294→   294→                >
   295→   295→                  <div className={`w-10 h-10 rounded-lg bg-gradient-to-br ${tool.color} flex items-center justify-center mb-3 group-hover:scale-110 transition-transform`}>
   296→   296→                    <tool.icon className="w-5 h-5 text-white" />
   297→   297→                  </div>
   298→   298→                  <h3 className="text-sm font-semibold text-white mb-1">{tool.label}</h3>
   299→   299→                  <p className="text-xs text-gray-400">{tool.description}</p>
   300→   300→                </button>
   301→   301→              ))}
   302→   302→            </div>
   303→   303→          </div>
   304→   304→        ) : (
   305→   305→          /* Active Tool Panel */
   306→   306→          <div className="flex-1 overflow-y-auto">
   307→   307→            {/* Back Button */}
   308→   308→            <button
   309→   309→              onClick={() => setActiveTool(null)}
   310→   310→              className="mx-4 mt-3 mb-2 flex items-center gap-2 text-sm text-gray-400 hover:text-white transition-colors"
   311→   311→            >
   312→   312→              ← Back to tools
   313→   313→            </button>
   314→   314→
   315→   315→            {/* Web Search */}
   316→   316→            {activeTool === 'web-search' && (
   317→   317→              <div className="px-4 pb-4 space-y-4">
   318→   318→                <div className="relative">
   319→   319→                  <input
   320→   320→                    type="text"
   321→   321→                    value={searchQuery}
   322→   322→                    onChange={(e) => setSearchQuery(e.target.value)}
   323→   323→                    onKeyDown={(e) => e.key === 'Enter' && handleWebSearch()}
   324→   324→                    placeholder="Search the web..."
   325→   325→                    className="w-full px-4 py-3 pl-11 bg-gray-800 border border-gray-700 rounded-xl text-white placeholder:text-gray-500 focus:border-cyan-500 focus:ring-1 focus:ring-cyan-500/20 outline-none"
   326→   326→                  />
   327→   327→                  <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   328→   328→                  <Button
   329→   329→                    onClick={handleWebSearch}
   330→   330→                    disabled={isSearching || !searchQuery.trim()}
   331→   331→                    size="sm"
   332→   332→                    className="absolute right-2 top-1/2 -translate-y-1/2 bg-gradient-to-r from-blue-500 to-cyan-500 hover:from-blue-400 hover:to-cyan-400"
   333→   333→                  >
   334→   334→                    {isSearching ? <Loader2 className="w-4 h-4 animate-spin" /> : 'Search'}
   335→   335→                  </Button>
   336→   336→                </div>
   337→   337→
   338→   338→                {searchResults.length > 0 && (
   339→   339→                  <div className="space-y-2">
   340→   340→                    <p className="text-sm text-gray-400">{searchResults.length} results found</p>
   341→   341→                    {searchResults.map((result, i) => (
   342→   342→                      <a
   343→   343→                        key={i}
   344→   344→                        href={result.url}
   345→   345→                        target="_blank"
   346→   346→                        rel="noopener noreferrer"
   347→   347→                        className="block p-3 rounded-lg bg-gray-800/50 hover:bg-gray-800 border border-gray-700/50 hover:border-cyan-500/30 transition-all group"
   348→   348→                      >
   349→   349→                        <div className="flex items-start justify-between gap-2">
   350→   350→                          <div className="flex-1 min-w-0">
   351→   351→                            <h4 className="text-sm font-medium text-white group-hover:text-cyan-400 truncate">{result.title}</h4>
   352→   352→                            <p className="text-xs text-gray-400 mt-1 line-clamp-2">{result.snippet}</p>
   353→   353→                            <span className="text-xs text-cyan-500/70 mt-1 inline-block">{result.source}</span>
   354→   354→                          </div>
   355→   355→                          <ExternalLink className="w-4 h-4 text-gray-500 group-hover:text-cyan-400 flex-shrink-0 mt-1" />
   356→   356→                        </div>
   357→   357→                      </a>
   358→   358→                    ))}
   359→   359→                  </div>
   360→   360→                )}
   361→   361→              </div>
   362→   362→            )}
   363→   363→
   364→   364→            {/* Image Generation */}
   365→   365→            {activeTool === 'image-gen' && (
   366→   366→              <div className="px-4 pb-4 space-y-4">
   367→   367→                <textarea
   368→   368→                  value={imagePrompt}
   369→   369→                  onChange={(e) => setImagePrompt(e.target.value)}
   370→   370→                  placeholder="Describe the image you want to generate..."
   371→   371→                  rows={3}
   372→   372→                  className="w-full px-4 py-3 bg-gray-800 border border-gray-700 rounded-xl text-white placeholder:text-gray-500 focus:border-purple-500 focus:ring-1 focus:ring-purple-500/20 outline-none resize-none"
   373→   373→                />
   374→   374→                
   375→   375→                <div className="flex gap-2">
   376→   376→                  {['512x512', '1024x1024', '1024x1792'].map(size => (
   377→   377→                    <button
   378→   378→                      key={size}
   379→   379→                      onClick={() => setImageSize(size)}
   380→   380→                      className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all ${
   381→   381→                        imageSize === size
   382→   382→                          ? 'bg-purple-500 text-white'
   383→   383→                          : 'bg-gray-800 text-gray-400 hover:bg-gray-700'
   384→   384→                      }`}
   385→   385→                    >
   386→   386→                      {size.split('x')[0]}
   387→   387→                    </button>
   388→   388→                  ))}
   389→   389→                </div>
   390→   390→
   391→   391→                <Button
   392→   392→                  onClick={handleGenerateImage}
   393→   393→                  disabled={isGeneratingImage || !imagePrompt.trim()}
   394→   394→                  className="w-full bg-gradient-to-r from-purple-500 to-pink-500 hover:from-purple-400 hover:to-pink-400"
   395→   395→                >
   396→   396→                  {isGeneratingImage ? (
   397→   397→                    <>
   398→   398→                      <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   399→   399→                      Generating...
   400→   400→                    </>
   401→   401→                  ) : (
   402→   402→                    <>
   403→   403→                      <Image className="w-4 h-4 mr-2" />
   404→   404→                      Generate Image
   405→   405→                    </>
   406→   406→                  )}
   407→   407→                </Button>
   408→   408→
   409→   409→                {generatedImage && (
   410→   410→                  <div className="mt-4">
   411→   411→                    <img src={generatedImage} alt="Generated" className="w-full rounded-xl" />
   412→   412→                    <div className="flex gap-2 mt-3">
   413→   413→                      <Button variant="outline" size="sm" className="flex-1">
   414→   414→                        <Download className="w-4 h-4 mr-2" /> Download
   415→   415→                      </Button>
   416→   416→                      <Button 
   417→   417→                        size="sm" 
   418→   418→                        className="flex-1 bg-gradient-to-r from-cyan-500 to-violet-500"
   419→   419→                        onClick={() => onInsertToChat(`[Generated Image: ${imagePrompt}]`)}
   420→   420→                      >
   421→   421→                        <Send className="w-4 h-4 mr-2" /> Insert to Chat
   422→   422→                      </Button>
   423→   423→                    </div>
   424→   424→                  </div>
   425→   425→                )}
   426→   426→              </div>
   427→   427→            )}
   428→   428→
   429→   429→            {/* Voice Chat */}
   430→   430→            {activeTool === 'voice-chat' && (
   431→   431→              <div className="px-4 pb-4 space-y-4">
   432→   432→                <textarea
   433→   433→                  value={voiceText}
   434→   434→                  onChange={(e) => setVoiceText(e.target.value)}
   435→   435→                  placeholder="Type or speak your message..."
   436→   436→                  rows={3}
   437→   437→                  className="w-full px-4 py-3 bg-gray-800 border border-gray-700 rounded-xl text-white placeholder:text-gray-500 focus:border-green-500 focus:ring-1 focus:ring-green-500/20 outline-none resize-none"
   438→   438→                />
   439→   439→
   440→   440→                <div className="flex gap-2">
   441→   441→                  <Button
   442→   442→                    onClick={toggleRecording}
   443→   443→                    variant={isRecording ? "destructive" : "outline"}
   444→   444→                    className={`flex-1 ${isRecording ? 'animate-pulse' : ''}`}
   445→   445→                  >
